  using System.Diagnostics;

    private async Task<int> BuildSetupProjectAsync(
        Application app, string outputFolder, CancellationToken cancellation)
    {
        var setupProjectFile = GetConfiguredPath(app.SetupProjectPath);
        if (string.IsNullOrWhiteSpace(setupProjectFile) || !File.Exists(setupProjectFile))
        {
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] Setup project not found: {app.SetupProjectPath}\r\n";
            return -1;
        }

        var setupProjectDir = Path.GetDirectoryName(setupProjectFile)!;
        var setupProjectName = Path.GetFileNameWithoutExtension(setupProjectFile);

        string? setupSolutionPath = null;
        if (!string.IsNullOrWhiteSpace(app.SetupSolutionPath))
            setupSolutionPath = GetConfiguredPath(app.SetupSolutionPath);

        if (string.IsNullOrWhiteSpace(setupSolutionPath) || !File.Exists(setupSolutionPath))
        {
            var parent = Path.GetDirectoryName(setupProjectDir);
            var candidate = parent is null ? null : Path.Combine(parent, "WebToClientBridge.sln");
            if (!string.IsNullOrEmpty(candidate) && File.Exists(candidate))
                setupSolutionPath = candidate;
            else if (!string.IsNullOrEmpty(parent))
                setupSolutionPath = Directory.GetFiles(parent, "*.sln").FirstOrDefault();
        }

        if (string.IsNullOrWhiteSpace(setupSolutionPath) || !File.Exists(setupSolutionPath))
        {
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] Setup solution (.sln) not found for {app.Name}\r\n";
            return -1;
        }

        var devenvPath = BuildService.ResolveVisualStudioIdePath();
        if (string.IsNullOrWhiteSpace(devenvPath) || !File.Exists(devenvPath))
        {
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] devenv not found.\r\n";
            return -1;
        }

        // Prefer console host for captured stdout
        var devenvCom = Path.ChangeExtension(devenvPath, ".com");
        if (File.Exists(devenvCom))
            devenvPath = devenvCom;

        // NO cmd.exe — FileName + Arguments avoids Program Files quoting bugs
        var args =
            $"\"{setupSolutionPath}\" /build Release /project \"{setupProjectName}\" /projectconfig Release";

        LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Building setup MSI for {app.Name}...\r\n";
        LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Setup exe: {devenvPath}\r\n";
        LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Setup args: {args}\r\n";

        var solutionDir = Path.GetDirectoryName(setupSolutionPath) ?? setupProjectDir;
        var exit = await RunProcessCaptureAsync(
            devenvPath,
            args,
            solutionDir,
            line =>
            {
                lock (_pendingLogBuffer)
                {
                    _pendingLogBuffer.Append(line);
                    if (!line.EndsWith('\n') && !line.EndsWith('\r'))
                        _pendingLogBuffer.AppendLine();
                }
                app.LogEntries.Add(line.TrimEnd('\r', '\n'));
            },
            cancellation);

        FlushPendingLogs();

        if (exit != 0)
        {
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] Setup build failed for {app.Name} with exit code {exit}\r\n";
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] If you still see devenv HELP text, this method was not used (still on RunCommandAsync/cmd).\r\n";
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] If no help text: install VS extension 'Installer Projects' and Build Web2ClientBridge.Setup once in IDE.\r\n";
            return exit;
        }

        var msiFiles = Directory.GetFiles(setupProjectDir, "*.msi", SearchOption.AllDirectories)
            .OrderByDescending(File.GetLastWriteTimeUtc)
            .ToArray();

        if (msiFiles.Length == 0)
        {
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] No .msi under {setupProjectDir} after setup build.\r\n";
            return -1;
        }

        Directory.CreateDirectory(outputFolder);
        foreach (var msi in msiFiles)
        {
            var dest = Path.Combine(outputFolder, Path.GetFileName(msi));
            File.Copy(msi, dest, overwrite: true);
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Setup MSI copied to {dest}\r\n";
        }

        return 0;
    }

    /// <summary>
    /// Runs an exe directly (no cmd.exe /c). Required for devenv under Program Files.
    /// </summary>
    private static async Task<int> RunProcessCaptureAsync(
        string fileName,
        string arguments,
        string workingDirectory,
        Action<string> onLine,
        CancellationToken cancellation)
    {
        var psi = new ProcessStartInfo
        {
            FileName = fileName,
            Arguments = arguments,
            WorkingDirectory = workingDirectory,
            UseShellExecute = false,
            RedirectStandardOutput = true,
            RedirectStandardError = true,
            CreateNoWindow = true
        };

        using var process = new Process { StartInfo = psi, EnableRaisingEvents = true };
        var tcs = new TaskCompletionSource<int>(TaskCreationOptions.RunContinuationsAsynchronously);

        process.OutputDataReceived += (_, e) =>
        {
            if (e.Data is not null)
                onLine(e.Data);
        };
        process.ErrorDataReceived += (_, e) =>
        {
            if (e.Data is not null)
                onLine(e.Data);
        };
        process.Exited += (_, _) => tcs.TrySetResult(process.ExitCode);

        if (!process.Start())
            return -1;

        process.BeginOutputReadLine();
        process.BeginErrorReadLine();

        await using (cancellation.Register(() =>
        {
            try
            {
                if (!process.HasExited)
                    process.Kill(entireProcessTree: true);
            }
            catch { /* ignore */ }
            tcs.TrySetCanceled(cancellation);
        }))
        {
            return await tcs.Task.ConfigureAwait(false);
        }
    }
