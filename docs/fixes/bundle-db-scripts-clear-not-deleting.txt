WHY Directory.Delete OFTEN FAILS ON WINDOWS
-------------------------------------------
Directory.Delete(path, recursive: true) throws (and skips clear) when:

  1. Any file is LOCKED (open in SSMS / VS / Notepad / Explorer preview)
  2. File has ReadOnly attribute
  3. Antivirus briefly locks the .sql
  4. Another process holds a handle on the folder

If it throws, your OUTER catch logs:
  [ERROR] Bundle Scripts Failed: ... Access denied / file in use ...
and nothing is cleared.

Even when Delete "works", the copy loop immediately recreates files —
Explorer will show files again. That is expected AFTER a successful clear+copy.
Stale = same OLD file still there with OLD timestamp / name not in this run.

            // Robust clear — Directory.Delete(recursive) often fails on locked/readonly files
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Clearing script folder: {destDir}\r\n";
            await InvokeAsync(StateHasChanged);

            if (Directory.Exists(destDir))
            {
                var existing = Directory.GetFiles(destDir, "*.*", SearchOption.AllDirectories);
                LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Found {existing.Length} existing file(s) to delete\r\n";

                var deleted = 0;
                var failed = 0;
                foreach (var oldFile in existing)
                {
                    try
                    {
                        // Clear ReadOnly so File.Delete can succeed
                        var attrs = File.GetAttributes(oldFile);
                        if ((attrs & FileAttributes.ReadOnly) != 0)
                            File.SetAttributes(oldFile, attrs & ~FileAttributes.ReadOnly);

                        File.Delete(oldFile);
                        deleted++;
                        LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Deleted: {oldFile}\r\n";
                    }
                    catch (Exception delEx)
                    {
                        failed++;
                        LogOutput += $"[{DateTime.Now:HH:mm:ss}] [WARN] Could not delete {oldFile}: {delEx.Message}\r\n";
                    }
                }

                // Remove empty subfolders (deepest first)
                foreach (var sub in Directory.GetDirectories(destDir, "*", SearchOption.AllDirectories)
                             .OrderByDescending(d => d.Length))
                {
                    try
                    {
                        if (!Directory.EnumerateFileSystemEntries(sub).Any())
                            Directory.Delete(sub, recursive: false);
                    }
                    catch { /* best-effort */ }
                }

                // VERIFY — if anything remains, STOP (do not copy on top of stale files)
                var remaining = Directory.Exists(destDir)
                    ? Directory.GetFiles(destDir, "*.*", SearchOption.AllDirectories)
                    : Array.Empty<string>();

                LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Clear result: deleted={deleted}, failed={failed}, remaining={remaining.Length}\r\n";

                if (remaining.Length > 0)
                {
                    LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] Could not clear script folder. Close files in Explorer/SSMS/VS and retry.\r\n";
                    foreach (var r in remaining)
                        LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] Still locked: {r}\r\n";
                    await InvokeAsync(StateHasChanged);
                    return;
                }
            }

            Directory.CreateDirectory(destDir);
            var copied = 0;
            foreach (var file in scriptFiles)
            {
                var relative = Path.GetRelativePath(builderOutputDir, file);
                var dest = Path.Combine(destDir, relative);
                Directory.CreateDirectory(Path.GetDirectoryName(dest)!);
                File.Copy(file, dest, overwrite: true);
                copied++;
                LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Copied {relative}\r\n";
            }

    private int ClearFolderContents(string folder, Action<string> log)
    {
        if (string.IsNullOrWhiteSpace(folder) || !Directory.Exists(folder))
            return 0;

        var deleted = 0;
        foreach (var oldFile in Directory.GetFiles(folder, "*.*", SearchOption.AllDirectories))
        {
            var attrs = File.GetAttributes(oldFile);
            if ((attrs & FileAttributes.ReadOnly) != 0)
                File.SetAttributes(oldFile, attrs & ~FileAttributes.ReadOnly);
            File.Delete(oldFile);
            deleted++;
            log($"Deleted: {oldFile}");
        }

        foreach (var sub in Directory.GetDirectories(folder, "*", SearchOption.AllDirectories)
                     .OrderByDescending(d => d.Length))
        {
            try
            {
                if (!Directory.EnumerateFileSystemEntries(sub).Any())
                    Directory.Delete(sub, recursive: false);
            }
            catch { }
        }

        return deleted;
    }
