3) Path helpers in BuildTab.razor

    private string MigrationShareRoot { get; set; } = @"\\EAGNMNSS595\Migration\GSS";
    private string ReleaseVersion { get; set; } = string.Empty;
    private string SprintNumber { get; set; } = string.Empty;

Load with other build settings from LoadAppSettings():

    MigrationShareRoot = settings.Build.MigrationShareRoot ?? MigrationShareRoot;
    ReleaseVersion = settings.Build.ReleaseVersion ?? ReleaseVersion;
    SprintNumber = settings.Build.SprintNumber ?? SprintNumber;

    private string GetSprintPackageRoot()
    {
        // \\EAGNMNSS595\Migration\GSS\8.8\Sprint 2
        return Path.Combine(
            MigrationShareRoot.TrimEnd('\\'),
            ReleaseVersion.Trim(),
            $"Sprint {SprintNumber.Trim()}");
    }

    private string GetServerApplicationDir() =>
        Path.Combine(GetSprintPackageRoot(), "server_application");

    private string GetServerDatabaseDir() =>
        Path.Combine(GetSprintPackageRoot(), "server_database");

    private string GetServerWebDir() =>
        Path.Combine(GetSprintPackageRoot(), "server_web");

    private void EnsureMigrationFolders()
    {
        Directory.CreateDirectory(GetServerApplicationDir());
        Directory.CreateDirectory(GetServerDatabaseDir());
        Directory.CreateDirectory(GetServerWebDir());
    }


--------------------------------------------------------------------------------
4) After successful build — copy into migration folders
--------------------------------------------------------------------------------
Call AFTER artifacts are in OutputRoot (and after optional _v1 archive of local output).

What goes where:

  Win Service / MSI / setup.exe  →  server_application\
  DB scripts (db-scripts)        →  server_database\
  Web publish (GSSWeb, ASMX…)    →  server_web\{appName}\

Example method:

    private async Task CopyBuildToMigrationShareAsync(Application app, string outputFolder)
    {
        if (string.IsNullOrWhiteSpace(ReleaseVersion) || string.IsNullOrWhiteSpace(SprintNumber))
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Set build:releaseVersion and build:sprintNumber in appsettings.");
            return;
        }

        EnsureMigrationFolders();

        var type = app.Type ?? string.Empty;
        string dest;

        if (type.Contains("WinService", StringComparison.OrdinalIgnoreCase)
            || type.Contains("Msi", StringComparison.OrdinalIgnoreCase))
        {
            dest = GetServerApplicationDir();
            // Copy setup.exe / .msi from outputFolder → dest
        }
        else
        {
            // Web / ASMX / MVC → server_web\{outputSubfolder}
            dest = Path.Combine(GetServerWebDir(), app.OutputSubfolder ?? app.Name);
            Directory.CreateDirectory(dest);
            // Copy published files from outputFolder → dest
        }

        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Copied {app.Name} to {dest}");
        await InvokeAsync(StateHasChanged);
    }

After Bundle Scripts succeeds, also copy:

    {outputRoot}\db-scripts\  →  GetServerDatabaseDir()
