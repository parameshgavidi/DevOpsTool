GOAL

Current (flat):

  F:\build-output\
    GSSApi\
    IMCApi\
    GSSMailerWebService\

New (version folder from branch, then same app folders):

      IMCApi\
      GSSMailerWebService\
      GSSWebService\
      db-scripts\           ← optional same rule for Bundle Scripts

Matches Migration share style: \\...\Migration\GSS\8.7\ ...


RULES
-----
1. Read current branch at Code Repo Root (after Get latest).
2. Trim remote prefix (origin/) and trim leading "GSSv" (or "GSS").
3. If version folder does not exist under Output Root → create it.
4. If it already exists → reuse (do not fail).
5. Publish each app into:  {outputRoot}\{version}\{outputSubfolder}\
6. If branch cannot be resolved → FALL BACK to old flat path
   {outputRoot}\{outputSubfolder}\  so current process still works.

    /// <summary>
    /// Current branch at RepoRoot, e.g. "GSSv8.7" or "origin/GSSv8.7".
    /// </summary>
    private string? TryGetCurrentGitBranch()
    {
        try
        {
            var root = RepoRoot?.Trim();
            if (string.IsNullOrWhiteSpace(root) || !Directory.Exists(root))
                return null;

            var psi = new System.Diagnostics.ProcessStartInfo
            {
                FileName = "cmd.exe",
                Arguments = "/c git rev-parse --abbrev-ref HEAD",
                WorkingDirectory = root,
                RedirectStandardOutput = true,
                RedirectStandardError = true,
                UseShellExecute = false,
                CreateNoWindow = true
            };

            using var p = System.Diagnostics.Process.Start(psi);
            if (p is null) return null;
            var stdout = p.StandardOutput.ReadToEnd().Trim();
            p.WaitForExit(15000);
            if (p.ExitCode != 0 || string.IsNullOrWhiteSpace(stdout) || stdout == "HEAD")
                return null;
            return stdout;
        }
        catch
        {
            return null;
        }
    }

    /// <summary>
    /// GSSv8.7 / origin/GSSv8.7 / GSS8.7 → "8.7"
    /// </summary>
    private static string? TrimGssVersionFolder(string? branchName)
    {
        if (string.IsNullOrWhiteSpace(branchName))
            return null;

        var name = branchName.Trim();

        // origin/GSSv8.7 → GSSv8.7
        var slash = name.LastIndexOf('/');
        if (slash >= 0 && slash < name.Length - 1)
            name = name[(slash + 1)..];

        // GSSv8.7 → 8.7
        if (name.StartsWith("GSSv", StringComparison.OrdinalIgnoreCase))
            return name[4..];

        // GSS8.7 → 8.7 (rare, no "v")
        if (name.StartsWith("GSS", StringComparison.OrdinalIgnoreCase)
            && name.Length > 3
            && char.IsDigit(name[3]))
            return name[3..];

        return null;
    }

    /// <summary>
    /// New: OutputRoot\8.7\GSSApi
    /// Fallback: OutputRoot\GSSApi  (current behavior)
    /// </summary>
    private string ResolveAppOutputFolder(Application app)
    {
        var outputRoot = (OutputRoot ?? string.Empty).Trim();
        var sub = (app.OutputSubfolder ?? app.Name ?? "app").Trim();

        var version = TrimGssVersionFolder(TryGetCurrentGitBranch());
        if (!string.IsNullOrWhiteSpace(version))
        {
            var versionDir = Path.Combine(outputRoot, version);
            Directory.CreateDirectory(versionDir);   // no-op if exists
            var folder = Path.Combine(versionDir, sub);
            Directory.CreateDirectory(folder);
            return folder;
        }

        // FALLBACK — do not break current flat layout
        var flat = Path.Combine(outputRoot, sub);
        Directory.CreateDirectory(flat);
        return flat;
    }
