================================================================================
Git Latest before build — per-project gitLatestCommand
Separate method GetLatestFromRepoAsync, called from BuildApplication
================================================================================

FLOW
----
1. Build selected app (or Build All) → BuildApplication
2. BuildApplication calls GetLatestFromRepoAsync(app, ...)
3. For THAT app only:
   a) fetch remote
   b) revert local changes under the project folder (tracked + untracked)
   c) restore that project folder from upstream (project-wise — NOT full-repo pull)
4. On success (true) → RestorePackagesAsync → msbuild / publish
5. On failure (false) → stop that app build (Status = Failed), log error


--------------------------------------------------------------------------------
1) appsettings.json — project-wise gitLatestCommand on EACH app
--------------------------------------------------------------------------------
Do NOT use plain "git pull --ff-only" — that updates the entire repo (all projects).

Each app uses its own project folder path:
  - GSS Web Service → Web Services\GSSWebService
  - GSSWeb         → WebV3\gssweb

Command steps (one string, chained with &&):
  1. git fetch                         — get remote updates
  2. git clean -fd -- "<projectDir>"   — remove untracked local files in project
  3. git restore --source=@{u} ...     — discard local edits + take upstream
                                         files for THIS project folder only

{
  "name": "GSS Web Service",
  "type": "AsmxWebService",
  "projectPath": "Web Services\\GSSWebService\\GSSWebService.sln",
  "gitLatestCommand": "git fetch && git clean -fd -- \"Web Services\\GSSWebService\" && git restore --source=@{u} --staged --worktree -- \"Web Services\\GSSWebService\"",
  "buildCommand": "msbuild \"{projectFile}\" /p:Configuration=Release /p:DeployOnBuild=true /p:PublishProfile=FolderProfile",
  "outputSubfolder": "GSSWebService"
},
{
  "name": "GSSWeb",
  "type": "AspNetWebForms",
  "projectPath": "WebV3\\gssweb\\gssweb.sln",
  "gitLatestCommand": "git fetch && git clean -fd -- \"WebV3\\gssweb\" && git restore --source=@{u} --staged --worktree -- \"WebV3\\gssweb\"",
  "buildCommand": "msbuild \"{projectFile}\" /p:Configuration=Release /p:DeployOnBuild=true /p:PublishProfile=FolderProfile",
  "outputSubfolder": "GSSWeb"
}

Notes:
- cwd must be Code Repo Root (build:repoRoot, e.g. F:\GSS\AppDev\gss-root) so .git is found.
- @{u} = current branch upstream (e.g. origin/GSSv9.2S2). Branch must have upstream set.
- This does NOT change other project folders under the same repo.
- Local changes under the project folder are discarded (by design) before taking latest.
- Do NOT put this inside buildCommand — GetLatestFromRepoAsync runs it first.
- If @{u} fails (no upstream), set upstream once:  git branch -u origin/<branch>
  Or temporarily use:  --source=origin/<branch>  instead of @{u}.

WARNING — NuGet Install-Package vs git latest:
  Local "Install-Package ..." changes (packages.config / *.csproj) WILL BE WIPED
  by git restore on the next build. That is expected.
  Do NOT put package names in appsettings.
  Fix: commit + push the package change to the GSS remote branch (so @{u}
  already has it), OR run nuget restore AFTER GetLatestFromRepoAsync.
  See: docs/fixes/gssweb-nuget-vs-git-latest.txt

3) BuildTab.razor — NEW method GetLatestFromRepoAsync
--------------------------------------------------------------------------------
Add this method (same style as RestorePackagesAsync). Returns true on success
so BuildApplication can abort when false.

private async Task<bool> GetLatestFromRepoAsync(Application app, string workingDir, CancellationToken cancellation)
{
    if (string.IsNullOrWhiteSpace(app.GitLatestCommand))
    {
        LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] No gitLatestCommand for {app.Name} — skipping project git latest\r\n";
        return true;
    }

    var gitCmd = app.GitLatestCommand.Trim();
    // Run from Code Repo Root so git sees .git (project paths in the command are relative to repo root)
    var gitWorkingDir = string.IsNullOrWhiteSpace(RepoRoot) ? workingDir : RepoRoot.Trim();

    LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Getting latest for {app.Name} (project-wise: revert local + restore from upstream)...\r\n";
    LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Git: {gitCmd}  (cwd: {gitWorkingDir})\r\n";
    await InvokeAsync(StateHasChanged);

    var gitExit = await BuildService.RunCommandAsync(
        gitWorkingDir,
        gitCmd,
        line =>
        {
            lock (_pendingLogBuffer)
            {
                _pendingLogBuffer.Append(line);
                if (!line.EndsWith('\n') && !line.EndsWith('\r'))
                    _pendingLogBuffer.AppendLine();
            }
        },
        cancellation);

    FlushPendingLogs();

    if (gitExit != 0)
    {
        LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] Git latest failed for {app.Name} with exit code {gitExit}\r\n";
        await InvokeAsync(StateHasChanged);
        return false;
    }

    LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Git latest OK for {app.Name}\r\n";
    await InvokeAsync(StateHasChanged);
    return true;
}
