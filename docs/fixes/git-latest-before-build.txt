Git Latest before build — per-project gitLatestCommand
Separate method GetLatestFromRepoAsync, called from BuildApplication
================================================================================

FLOW
----
1. Build selected app (or Build All) → BuildApplication
2. BuildApplication calls GetLatestFromRepoAsync(app, ...)
3. On success (true) → RestorePackagesAsync → msbuild / publish
4. On failure (false) → stop that app build (Status = Failed), log error

Notes:
- Same command can be used for every project when they share one Code Repo Root
  (build:repoRoot, e.g. F:\GSS\AppDev\gss-root).
- Per-app override is allowed (different branch, remote, or repo path later).
- Do NOT put git pull inside buildCommand — run it as its own method first.
- Prefer:  git pull --ff-only

3) BuildTab.razor — NEW method GetLatestFromRepoAsync
--------------------------------------------------------------------------------
Add this method (same style as RestorePackagesAsync). Returns true on success
so BuildApplication can abort when false.

private async Task<bool> GetLatestFromRepoAsync(Application app, string workingDir, CancellationToken cancellation)
{
    if (string.IsNullOrWhiteSpace(app.GitLatestCommand))
    {
        LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] No gitLatestCommand for {app.Name} — skipping git pull\r\n";
        return true;
    }

    var gitCmd = app.GitLatestCommand.Trim();
    // Run from Code Repo Root so git sees .git (not the project subfolder)
    var gitWorkingDir = string.IsNullOrWhiteSpace(RepoRoot) ? workingDir : RepoRoot.Trim();

    LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Getting latest for {app.Name}...\r\n";
    LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Git: {gitCmd}  (cwd: {gitWorkingDir})\r\n";
    await InvokeAsync(StateHasChanged);

    var gitExit = await BuildService.RunCommandAsync(
        gitWorkingDir,
        gitCmd,
        line =>
        {
            lock (_pendingLogBuffer)
            {
                _pendingLogBuffer.Append(line);
                if (!line.EndsWith('\n') && !line.EndsWith('\r'))
                    _pendingLogBuffer.AppendLine();
            }
        },
        cancellation);

    FlushPendingLogs();

    if (gitExit != 0)
    {
        LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] Git latest failed for {app.Name} with exit code {gitExit}\r\n";
        await InvokeAsync(StateHasChanged);
        return false;
    }

    LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Git latest OK for {app.Name}\r\n";
    await InvokeAsync(StateHasChanged);
    return true;
}
