    private async Task LoadDbBranchesAsync()
    {
        if (_isLoadingBranches)
            return;

        _isLoadingBranches = true;
        try
        {
            // SAME as previous: list branches from Code Repo Root (gss-root)
            var root = RepoRoot?.Trim() ?? string.Empty;
            if (string.IsNullOrWhiteSpace(root) || !Directory.Exists(root))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Repo Root not found: {root}. Set Repo Root first.");
                await InvokeAsync(StateHasChanged);
                return;
            }

            // Previous command — hardcoded, cmd-safe (NO --format quotes)
            var command = "git branch -r";
            var lines = new List<string>();

            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Loading DB branches (git) from {root}...");
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Command: {command}");
            await InvokeAsync(StateHasChanged);

            var exit = await BuildService.RunCommandAsync(
                root,
                command,
                line =>
                {
                    foreach (var raw in SplitBranchTokens(line))
                    {
                        var t = NormalizeBranchName(raw);
                        if (string.IsNullOrWhiteSpace(t))
                            continue;
                        // ONLY GSS release branches (drops obj / Scripts / cmd junk)
                        if (!t.StartsWith("GSS", StringComparison.OrdinalIgnoreCase))
                            continue;
                        if (!lines.Contains(t, StringComparer.OrdinalIgnoreCase))
                            lines.Add(t);
                    }
                },
                CancellationToken.None);

            if (exit != 0)
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Failed to list branches (exit {exit})");
                await InvokeAsync(StateHasChanged);
                return;
            }

            lines.Sort(StringComparer.OrdinalIgnoreCase);
            DbBranchOptions = lines;

            if (DbBranchOptions.Count == 0)
            {
                DbRollupToBranch = string.Empty;
                DbRollupFromBranch = string.Empty;
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Parsed 0 GSS branches. Check Repo Root is a Git repo.");
            }
            else
            {
                // Default To = LATEST branch (last after sort)
                var latest = DbBranchOptions[^1];
                DbRollupToBranch = latest;
                DbRollupFromBranch = GetPreviousBranch(DbBranchOptions, latest) ?? string.Empty;

                AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Loaded {DbBranchOptions.Count} branch(es). To={DbRollupToBranch}, From={DbRollupFromBranch}");
                var preview = Math.Min(8, DbBranchOptions.Count);
                for (var i = 0; i < preview; i++)
                    AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO]   [{i}] {DbBranchOptions[i]}");
            }

            await InvokeAsync(StateHasChanged);
        }
        catch (Exception ex)
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Load branches failed: {ex.Message}");
            await InvokeAsync(StateHasChanged);
        }
        finally
        {
            _isLoadingBranches = false;
            await InvokeAsync(StateHasChanged);
        }
    }

    private Task OnToBranchChanged(ChangeEventArgs e)
    {
        var to = e.Value?.ToString()?.Trim() ?? string.Empty;
        DbRollupToBranch = to;
        DbRollupFromBranch = GetPreviousBranch(DbBranchOptions, to) ?? string.Empty;
        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] To Branch={DbRollupToBranch} → From Branch={DbRollupFromBranch}");
        return InvokeAsync(StateHasChanged);
    }

    private static string? GetPreviousBranch(List<string> options, string toBranch)
    {
        if (options.Count == 0 || string.IsNullOrWhiteSpace(toBranch))
            return null;

        var idx = options.FindIndex(x =>
            string.Equals(x, toBranch, StringComparison.OrdinalIgnoreCase));
        if (idx <= 0)
            return null;

        return options[idx - 1];
    }

    private static IEnumerable<string> SplitBranchTokens(string? line)
    {
        var s = (line ?? string.Empty).Trim();
        if (string.IsNullOrWhiteSpace(s))
            yield break;
        if (s.StartsWith(">>>", StringComparison.Ordinal))
            yield break;

        if (s.Contains('\n') || s.Contains('\r'))
        {
            foreach (var part in s.Split(new[] { '\r', '\n' }, StringSplitOptions.RemoveEmptyEntries))
            {
                foreach (var token in SplitBranchTokens(part))
                    yield return token;
            }
            yield break;
        }

        if (s.Contains('\'') || s.Contains('"'))
        {
            foreach (var p in s.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries))
                yield return p;
            yield break;
        }

        yield return s;
    }

    private static string NormalizeBranchName(string raw)
    {
        var t = (raw ?? string.Empty).Trim();
        if (string.IsNullOrWhiteSpace(t))
            return string.Empty;

        if (t.StartsWith("*"))
            t = t.TrimStart('*').Trim();

        t = t.Trim().Trim('\'', '"').Trim();

        if (string.IsNullOrWhiteSpace(t))
            return string.Empty;
        if (string.Equals(t, "origin", StringComparison.OrdinalIgnoreCase))
            return string.Empty;
        if (t.Contains("HEAD", StringComparison.OrdinalIgnoreCase))
            return string.Empty;

        while (t.StartsWith("origin/", StringComparison.OrdinalIgnoreCase))
            t = t.Substring("origin/".Length).Trim().Trim('\'', '"').Trim();

        return t.Trim();
    }
