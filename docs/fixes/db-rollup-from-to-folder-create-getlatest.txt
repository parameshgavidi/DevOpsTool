    private async Task EnsureFromAndToBranchFoldersAsync()
    {
        if (!IsFromLessThanTo())
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] From Branch must be less than To Branch. From={DbRollupFromBranch}, To={DbRollupToBranch}");
            await InvokeAsync(StateHasChanged);
            return;
        }

        // Separate folders — From then To
        if (!string.IsNullOrWhiteSpace(DbRollupFromBranch))
            await EnsureDbBranchFolderAsync(DbRollupFromBranch);

        if (!string.IsNullOrWhiteSpace(DbRollupToBranch)
            && !string.Equals(DbRollupToBranch, DbRollupFromBranch, StringComparison.OrdinalIgnoreCase))
            await EnsureDbBranchFolderAsync(DbRollupToBranch);

        await InvokeAsync(StateHasChanged);
    }

    /// <summary>
    /// Under Script Source: if {branch} folder missing → create + get latest;
    /// if it exists → get latest only.
    /// Example: F:\GSS\DBRollupScriptBuilder\GSSv9.2S1
    /// </summary>
    private async Task EnsureDbBranchFolderAsync(string branch)
    {
        var name = branch?.Trim() ?? string.Empty;
        if (string.IsNullOrWhiteSpace(name))
            return;

        var scriptRoot = ScriptSource?.Trim() ?? string.Empty;
        if (string.IsNullOrWhiteSpace(scriptRoot))
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Script Source is empty — cannot create/get branch folder for {name}");
            return;
        }

        if (!Directory.Exists(scriptRoot))
        {
            try
            {
                Directory.CreateDirectory(scriptRoot);
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Created Script Source: {scriptRoot}");
            }
            catch (Exception ex)
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Cannot create Script Source {scriptRoot}: {ex.Message}");
                return;
            }
        }

        var folder = Path.Combine(scriptRoot, name);
        var created = false;

        if (!Directory.Exists(folder))
        {
            Directory.CreateDirectory(folder);
            created = true;
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Created branch folder: {folder}");
        }
        else
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Branch folder exists — getting latest: {folder}");
        }

        await GetLatestIntoBranchFolderAsync(name, folder, created);
    }

    /// <summary>
    /// Get latest code for one branch into its Script Source folder.
    /// Prefers git worktree from RepoRoot (gss-root). Falls back to
    /// fetch/pull if the folder already has its own .git.
    /// </summary>
    private async Task GetLatestIntoBranchFolderAsync(string branch, string folder, bool justCreated)
    {
        var repoRoot = RepoRoot?.Trim() ?? string.Empty;
        if (string.IsNullOrWhiteSpace(repoRoot) || !Directory.Exists(repoRoot))
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [WARN] Repo Root missing — folder ready at {folder} but skipped git get-latest for {branch}");
            return;
        }

        var folderGit = Path.Combine(folder, ".git");
        var hasGit = Directory.Exists(folderGit) || File.Exists(folderGit);

        string command;
        string cwd;

        if (hasGit)
        {
            // Existing checkout/worktree → pull latest on that branch
            command = $"git fetch origin && git checkout {branch} && git pull --ff-only";
            cwd = folder;
        }
        else if (justCreated || !Directory.EnumerateFileSystemEntries(folder).Any())
        {
            // Empty / just-created folder → add as worktree of RepoRoot branch
            // (worktree fails if branch already checked out elsewhere — then clone)
            command =
                $"git fetch origin && " +
                $"(git worktree add \"{folder}\" \"{branch}\" || " +
                $"git clone --branch \"{branch}\" --single-branch \"{repoRoot}\" \"{folder}\")";
            cwd = repoRoot;

            // worktree add wants the target path to NOT exist as a normal dir.
            // If we already CreateDirectory'd an empty folder, remove it first.
            try
            {
                if (Directory.Exists(folder)
                    && !Directory.EnumerateFileSystemEntries(folder).Any())
                {
                    Directory.Delete(folder);
                }
            }
            catch (Exception ex)
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [WARN] Could not clear empty folder before worktree: {ex.Message}");
            }
        }
        else
        {
            // Folder has files but no .git — leave contents; do not wipe
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [WARN] Folder has files but no .git — skipped get-latest: {folder}");
            return;
        }

        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Get latest branch '{branch}'...");
        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Cwd: {cwd}");
        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Git: {command}");
        await InvokeAsync(StateHasChanged);

        var exit = await BuildService.RunCommandAsync(
            cwd,
            command,
            line =>
            {
                lock (_pendingLogBuffer)
                {
                    _pendingLogBuffer.Append(line);
                    if (!line.EndsWith('\n') && !line.EndsWith('\r'))
                        _pendingLogBuffer.AppendLine();
                }
            },
            CancellationToken.None);

        FlushPendingLogs();

        if (exit != 0)
        {
            // Ensure folder still exists even if git failed
            Directory.CreateDirectory(folder);
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Get latest failed for {branch} (exit {exit}). Folder kept at {folder}");
            return;
        }

        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Get latest OK for {branch} → {folder}");
    }

PASTE helpers:

    private bool IsFromLessThanTo()
    {
        var from = DbRollupFromBranch?.Trim() ?? string.Empty;
        var to = DbRollupToBranch?.Trim() ?? string.Empty;
        if (string.IsNullOrWhiteSpace(from) || string.IsNullOrWhiteSpace(to))
            return false;

        return CompareGssBranchNames(from, to) < 0;
    }

    /// <summary>
    /// GSSv9.1S1 &lt; GSSv9.1S2 &lt; GSSv9.2S1 &lt; GSSv9.2S2
    /// Strips GSS / GSSv prefix, compares version then sprint number.
    /// </summary>
    private static int CompareGssBranchNames(string a, string b)
    {
        var pa = ParseGssBranch(a);
        var pb = ParseGssBranch(b);
        if (pa is null && pb is null)
            return string.Compare(a, b, StringComparison.OrdinalIgnoreCase);
        if (pa is null) return -1;
        if (pb is null) return 1;

        var n = Math.Max(pa.VersionParts.Count, pb.VersionParts.Count);
        for (var i = 0; i < n; i++)
        {
            var x = i < pa.VersionParts.Count ? pa.VersionParts[i] : 0;
            var y = i < pb.VersionParts.Count ? pb.VersionParts[i] : 0;
            if (x != y) return x.CompareTo(y);
        }

        return pa.Sprint.CompareTo(pb.Sprint);
    }

    private sealed class GssBranchParts
    {
        public List<int> VersionParts { get; init; } = new();
        public int Sprint { get; init; }
    }

    private static GssBranchParts? ParseGssBranch(string name)
    {
        if (string.IsNullOrWhiteSpace(name))
            return null;

        var t = name.Trim();
        if (t.StartsWith("GSSv", StringComparison.OrdinalIgnoreCase))
            t = t.Substring(4);
        else if (t.StartsWith("GSS", StringComparison.OrdinalIgnoreCase))
            t = t.Substring(3);

        t = t.Trim();
        if (string.IsNullOrWhiteSpace(t))
            return null;

        // Split sprint: 9.2S1 → version=9.2, sprint=1
        var sprint = 0;
        var sIdx = t.LastIndexOf('S');
        if (sIdx < 0) sIdx = t.LastIndexOf('s');
        string versionText;
        if (sIdx > 0 && sIdx < t.Length - 1
            && int.TryParse(t[(sIdx + 1)..], out sprint))
        {
            versionText = t[..sIdx];
        }
        else
        {
            versionText = t;
            sprint = 0;
        }

        var parts = new List<int>();
        foreach (var p in versionText.Split('.', StringSplitOptions.RemoveEmptyEntries))
        {
            if (!int.TryParse(p, out var n))
                return null;
            parts.Add(n);
        }

        if (parts.Count == 0)
            return null;

        return new GssBranchParts { VersionParts = parts, Sprint = sprint };
    }
