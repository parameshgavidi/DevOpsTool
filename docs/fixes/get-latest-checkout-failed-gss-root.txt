PASTE this whole method (GET-LATEST-CHECKOUT-V2). It:

  A) Logs every git line immediately ([GIT] prefix) so fatal/error is visible
  B) Redirects stderr (2>&1) so cmd.exe errors reach RunCommandAsync
  C) Stashes local changes before switching (same idea as per-app gitLatestCommand)
  D) Uses  git checkout --ignore-other-worktrees -B <branch> origin/<branch>
     so a DB-rollup worktree cannot block gss-root
  E) On failure runs  git status -sb  +  git worktree list  into OUTPUT LOG

Keep GetLatestFromRepoAsync / LoadApplicationsFromConfigAsync unchanged.

    private async Task GetLatestAndShowAppsAsync()
    {
        if (_isGettingLatest)
            return;

        _isGettingLatest = true;
        _showApplications = false;
        try
        {
            var root = RepoRoot?.Trim() ?? string.Empty;
            if (string.IsNullOrWhiteSpace(root) || !Directory.Exists(root))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Repo Root not found: {root}");
                await InvokeAsync(StateHasChanged);
                return;
            }

            var gitDir = Path.Combine(root, ".git");
            if (!Directory.Exists(gitDir) && !File.Exists(gitDir))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Not a git repo (missing .git): {root}");
                await InvokeAsync(StateHasChanged);
                return;
            }

            var branch = RepoFromBranch?.Trim() ?? string.Empty;
            if (string.IsNullOrWhiteSpace(branch))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Select a branch in Get Latest From Repo Branch (click Refresh first).");
                await InvokeAsync(StateHasChanged);
                return;
            }

            // GET-LATEST-CHECKOUT-V2 — stash + fetch + force-track origin/<branch>
            var checkoutCmd =
                $"(git stash push -m \"devops-get-latest\" 2>&1 || ver > nul) && " +
                $"git fetch origin 2>&1 && " +
                $"git checkout --ignore-other-worktrees -B {branch} origin/{branch} 2>&1 && " +
                $"git pull --ff-only 2>&1";

            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] GET-LATEST-CHECKOUT-V2 Switching to branch '{branch}' at {root}...");
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Git: {checkoutCmd}");
            await InvokeAsync(StateHasChanged);

            StartLogFlushTimer();

            var checkoutExit = await RunGitLoggedAsync(root, checkoutCmd);

            if (checkoutExit != 0)
            {
                var trackCmd =
                    $"(git stash push -m \"devops-get-latest\" 2>&1 || ver > nul) && " +
                    $"git fetch origin 2>&1 && " +
                    $"git checkout --ignore-other-worktrees {branch} 2>&1";
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Checkout failed (exit {checkoutExit}). Trying local: {trackCmd}");
                await InvokeAsync(StateHasChanged);

                checkoutExit = await RunGitLoggedAsync(root, trackCmd);
            }

            if (checkoutExit != 0)
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Failed to checkout branch '{branch}' (exit {checkoutExit})");
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Diagnostic git status / worktrees:");
                await RunGitLoggedAsync(root, "git status -sb 2>&1 & git worktree list 2>&1 & git branch -vv 2>&1");
                await InvokeAsync(StateHasChanged);
                return;
            }

            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] On branch '{branch}' — getting latest per app...");
            await InvokeAsync(StateHasChanged);

            await LoadApplicationsFromConfigAsync();

            if (BuildApps.Count == 0)
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] No applications in appsettings (build:applications)");
                await InvokeAsync(StateHasChanged);
                return;
            }

            var okCount = 0;
            var failCount = 0;

            foreach (var app in BuildApps)
            {
                app.Status = "Getting latest...";
                await InvokeAsync(StateHasChanged);

                var ok = await GetLatestFromRepoAsync(app, root, CancellationToken.None);
                if (ok)
                {
                    app.Status = "Ready";
                    okCount++;
                }
                else
                {
                    app.Status = "Failed";
                    failCount++;
                }

                await InvokeAsync(StateHasChanged);
            }

            FlushPendingLogs();

            _showApplications = true;
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Get latest complete on '{branch}' — OK: {okCount}, Failed: {failCount}");
            await InvokeAsync(StateHasChanged);
        }
        catch (Exception ex)
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Get latest failed: {ex.Message}");
            _showApplications = false;
            await InvokeAsync(StateHasChanged);
        }
        finally
        {
            StopLogFlushTimer();
            _isGettingLatest = false;
            await InvokeAsync(StateHasChanged);
        }
    }

    /// <summary>
    /// Run git and print every stdout/stderr line immediately ([GIT]).
    /// Do NOT only buffer — that is why "Checkout failed" had no git text.
    /// </summary>
    private async Task<int> RunGitLoggedAsync(string workingDir, string command)
    {
        return await BuildService.RunCommandAsync(
            workingDir,
            command,
            line =>
            {
                var text = (line ?? string.Empty).TrimEnd('\r', '\n');
                if (text.Length == 0)
                    return;
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [GIT] {text}");
            },
            CancellationToken.None);
    }
