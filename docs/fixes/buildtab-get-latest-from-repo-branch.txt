    private async Task LoadRepoBranchesAsync()
    {
        if (_isLoadingRepoBranches)
            return;

        _isLoadingRepoBranches = true;
        try
        {
            var root = RepoRoot?.Trim() ?? string.Empty;
            if (string.IsNullOrWhiteSpace(root) || !Directory.Exists(root))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Repo Root not found: {root}. Browse Repo Root first.");
                await InvokeAsync(StateHasChanged);
                return;
            }

            // NO appsettings key — hardcoded cmd-safe command (same idea as From Branch)
            var command = "git branch -r";
            var lines = new List<string>();

            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Loading repo branches from {root}...");
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Command: {command}");
            await InvokeAsync(StateHasChanged);

            var exit = await BuildService.RunCommandAsync(
                root,
                command,
                line =>
                {
                    foreach (var raw in SplitBranchTokens(line))
                    {
                        var t = NormalizeBranchName(raw);
                        if (string.IsNullOrWhiteSpace(t))
                            continue;
                        // Keep only GSS release branches (drops cmd.exe / git / junk)
                        if (!t.StartsWith("GSS", StringComparison.OrdinalIgnoreCase))
                            continue;
                        if (!lines.Contains(t, StringComparer.OrdinalIgnoreCase))
                            lines.Add(t);
                    }
                },
                CancellationToken.None);

            if (exit != 0)
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Failed to list repo branches (exit {exit})");
                await InvokeAsync(StateHasChanged);
                return;
            }

            lines.Sort(StringComparer.OrdinalIgnoreCase);
            RepoBranchOptions = lines;

            if (RepoBranchOptions.Count == 0)
            {
                RepoFromBranch = string.Empty;
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Parsed 0 GSS branches. Check Repo Root is a Git repo.");
            }
            else
            {
                // Default = LATEST branch (last after sort, e.g. GSSv8.9S2)
                var latest = RepoBranchOptions[^1];
                if (string.IsNullOrWhiteSpace(RepoFromBranch)
                    || !RepoBranchOptions.Contains(RepoFromBranch, StringComparer.OrdinalIgnoreCase))
                {
                    RepoFromBranch = latest;
                }

                AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Loaded {RepoBranchOptions.Count} repo branch(es). Default latest = {RepoFromBranch}");
                var preview = Math.Min(8, RepoBranchOptions.Count);
                for (var i = 0; i < preview; i++)
                    AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO]   [{i}] {RepoBranchOptions[i]}");
            }

            await InvokeAsync(StateHasChanged);
        }
        catch (Exception ex)
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Load repo branches failed: {ex.Message}");
            await InvokeAsync(StateHasChanged);
        }
        finally
        {
            _isLoadingRepoBranches = false;
            await InvokeAsync(StateHasChanged);
        }
    }

    private static IEnumerable<string> SplitBranchTokens(string? line)
    {
        var s = (line ?? string.Empty).Trim();
        if (string.IsNullOrWhiteSpace(s))
            yield break;
        if (s.StartsWith(">>>", StringComparison.Ordinal))
            yield break;

        // BuildService may deliver one multi-line chunk — split on newlines first
        if (s.Contains('\n') || s.Contains('\r'))
        {
            foreach (var part in s.Split(new[] { '\r', '\n' }, StringSplitOptions.RemoveEmptyEntries))
            {
                foreach (var token in SplitBranchTokens(part))
                    yield return token;
            }
            yield break;
        }

        // Quoted blob: 'origin/A' 'origin/B'
        if (s.Contains('\'') || s.Contains('"'))
        {
            foreach (var p in s.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries))
                yield return p;
            yield break;
        }

        yield return s;
    }

    private static string NormalizeBranchName(string raw)
    {
        var t = (raw ?? string.Empty).Trim();
        if (string.IsNullOrWhiteSpace(t))
            return string.Empty;

        if (t.StartsWith("*"))
            t = t.TrimStart('*').Trim();

        t = t.Trim().Trim('\'', '"').Trim();

        if (string.IsNullOrWhiteSpace(t))
            return string.Empty;
        if (string.Equals(t, "origin", StringComparison.OrdinalIgnoreCase))
            return string.Empty;

        // Per-token HEAD check (do NOT use Contains on a multi-line blob)
        if (t.Contains("HEAD", StringComparison.OrdinalIgnoreCase))
            return string.Empty;

        while (t.StartsWith("origin/", StringComparison.OrdinalIgnoreCase))
            t = t.Substring("origin/".Length).Trim().Trim('\'', '"').Trim();

        return t.Trim();
    }

    private async Task GetLatestAndShowAppsAsync()
    {
        if (_isGettingLatest)
            return;

        _isGettingLatest = true;
        _showApplications = false;
        try
        {
            var root = RepoRoot?.Trim() ?? string.Empty;
            if (string.IsNullOrWhiteSpace(root) || !Directory.Exists(root))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Repo Root not found: {root}");
                await InvokeAsync(StateHasChanged);
                return;
            }

            var gitDir = Path.Combine(root, ".git");
            if (!Directory.Exists(gitDir) && !File.Exists(gitDir))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Not a git repo (missing .git): {root}");
                await InvokeAsync(StateHasChanged);
                return;
            }

            var branch = RepoFromBranch?.Trim() ?? string.Empty;
            if (string.IsNullOrWhiteSpace(branch))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Select a branch in Get Latest From Repo Branch (click Refresh first).");
                await InvokeAsync(StateHasChanged);
                return;
            }

            // A) Checkout selected branch at Repo Root (NO appsettings variable)
            var checkoutCmd =
                $"git fetch origin && git checkout {branch} && git pull --ff-only";

            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Switching to branch '{branch}' at {root}...");
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Git: {checkoutCmd}");
            await InvokeAsync(StateHasChanged);

            StartLogFlushTimer();

            var checkoutExit = await BuildService.RunCommandAsync(
                root,
                checkoutCmd,
                line =>
                {
                    lock (_pendingLogBuffer)
                    {
                        _pendingLogBuffer.Append(line);
                        if (!line.EndsWith('\n') && !line.EndsWith('\r'))
                            _pendingLogBuffer.AppendLine();
                    }
                },
                CancellationToken.None);

            FlushPendingLogs();

            if (checkoutExit != 0)
            {
                // Local branch missing — create tracking branch from origin/<branch>
                var trackCmd =
                    $"git fetch origin && git checkout -B {branch} origin/{branch}";
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Checkout failed (exit {checkoutExit}). Trying track: {trackCmd}");
                await InvokeAsync(StateHasChanged);

                checkoutExit = await BuildService.RunCommandAsync(
                    root,
                    trackCmd,
                    line =>
                    {
                        lock (_pendingLogBuffer)
                        {
                            _pendingLogBuffer.Append(line);
                            if (!line.EndsWith('\n') && !line.EndsWith('\r'))
                                _pendingLogBuffer.AppendLine();
                        }
                    },
                    CancellationToken.None);

                FlushPendingLogs();

                if (checkoutExit != 0)
                {
                    AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Failed to checkout branch '{branch}' (exit {checkoutExit})");
                    await InvokeAsync(StateHasChanged);
                    return;
                }
            }

            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] On branch '{branch}' — getting latest per app...");
            await InvokeAsync(StateHasChanged);

            // B) Load ALL apps from appsettings
            await LoadApplicationsFromConfigAsync();

            if (BuildApps.Count == 0)
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] No applications in appsettings (build:applications)");
                await InvokeAsync(StateHasChanged);
                return;
            }

            var okCount = 0;
            var failCount = 0;

            // C) Per-app get latest (uses each app's gitLatestCommand)
            foreach (var app in BuildApps)
            {
                app.Status = "Getting latest...";
                await InvokeAsync(StateHasChanged);

                var ok = await GetLatestFromRepoAsync(app, root, CancellationToken.None);
                if (ok)
                {
                    app.Status = "Ready";
                    okCount++;
                }
                else
                {
                    app.Status = "Failed";
                    failCount++;
                }

                await InvokeAsync(StateHasChanged);
            }

            FlushPendingLogs();

            // D) Show grid
            _showApplications = true;
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Get latest complete on '{branch}' — OK: {okCount}, Failed: {failCount}");
            await InvokeAsync(StateHasChanged);
        }
        catch (Exception ex)
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Get latest failed: {ex.Message}");
            _showApplications = false;
            await InvokeAsync(StateHasChanged);
        }
        finally
        {
            StopLogFlushTimer();
            _isGettingLatest = false;
            await InvokeAsync(StateHasChanged);
        }
    }
