4) LoadDbBranchesAsync — list branches from gss-database (optional but recommended)
--------------------------------------------------------------------------------
Runs: git branch -r   (or git for-each-ref) under DbRepoRoot.
Fills DbBranchOptions so dropdowns work like a selectable list.

    private async Task LoadDbBranchesAsync()
    {
        if (_isLoadingBranches)
            return;

        _isLoadingBranches = true;
        try
        {
            var root = DbRepoRoot?.Trim() ?? string.Empty;
            if (string.IsNullOrWhiteSpace(root) || !Directory.Exists(root))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] DB repo root not found: {root}. Set build:dbRepoRoot (gss-database).");
                await InvokeAsync(StateHasChanged);
                return;
            }

            // List remote branches; strip "origin/" prefix for -f / -t values
            var command = "git branch -r --format=\"%(refname:short)\"";
            var lines = new List<string>();

            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Loading DB branches from {root}...");
            await InvokeAsync(StateHasChanged);

            var exit = await BuildService.RunCommandAsync(
                root,
                command,
                line =>
                {
                    var t = (line ?? string.Empty).Trim();
                    if (string.IsNullOrWhiteSpace(t))
                        return;
                    if (t.Contains("HEAD", StringComparison.OrdinalIgnoreCase))
                        return;
                    // origin/GSSv9.2S2 → GSSv9.2S2
                    if (t.StartsWith("origin/", StringComparison.OrdinalIgnoreCase))
                        t = t.Substring("origin/".Length);
                    if (!lines.Contains(t, StringComparer.OrdinalIgnoreCase))
                        lines.Add(t);
                },
                CancellationToken.None);

            if (exit != 0)
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Failed to list branches (exit {exit})");
                await InvokeAsync(StateHasChanged);
                return;
            }

            lines.Sort(StringComparer.OrdinalIgnoreCase);
            DbBranchOptions = lines;

            // Keep current From/To if still in list; otherwise leave user-typed values
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Loaded {DbBranchOptions.Count} branch(es). Select From/To Branch, then Bundle Scripts.");
            await InvokeAsync(StateHasChanged);
        }
        catch (Exception ex)
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Load branches failed: {ex.Message}");
            await InvokeAsync(StateHasChanged);
        }
        finally
        {
            _isLoadingBranches = false;
            await InvokeAsync(StateHasChanged);
        }
    }
