ROOT CAUSE
----------
BuildService often delivers git stdout as ONE multi-line chunk (not one
callback per line). V5 did:

  1) treat whole chunk as one token
  2) strip first "origin/"
  3) if (t.Contains("HEAD")) return "";   ← KILLS EVERYTHING

Because the chunk also contains "origin/HEAD -> origin/main", Contains("HEAD")
is true → Normalize returns empty → parsed=0 → dropdown empty.

Raw lines still appear in the log (one AppendLog with embedded newlines),
so it looks like branches "loaded" but nothing binds to DbBranchOptions.

    private async Task LoadDbBranchesAsync()

            return;

        _isLoadingBranches = true;

            if (exit != 0)

                DbBranchOptions = new List<string>();
                await InvokeAsync(StateHasChanged);

            var gss = lines

                .ToList();
            if (gss.Count > 0)
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] GSS* filter: {gss.Count} of {lines.Count}");
                lines = gss;

            DbBranchOptions = lines;

            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] BRANCH-LOAD-V6 dropdown count={DbBranchOptions.Count}");
            for (var i = 0; i < Math.Min(8, DbBranchOptions.Count); i++)
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO]   [{i}] {DbBranchOptions[i]}");

            if (DbBranchOptions.Count > 0)
            {
                var fromCfg = settings?.Build?.DbRollupFromBranch?.Trim();
                var toCfg = settings?.Build?.DbRollupToBranch?.Trim();

                DbRollupFromBranch = !string.IsNullOrWhiteSpace(fromCfg)
                    && DbBranchOptions.Contains(fromCfg, StringComparer.OrdinalIgnoreCase)
                        ? fromCfg!
                        : DbBranchOptions[0];

                DbRollupToBranch = !string.IsNullOrWhiteSpace(toCfg)
                    && DbBranchOptions.Contains(toCfg, StringComparer.OrdinalIgnoreCase)

    // Split a callback chunk into individual branch lines/tokens
    private static IEnumerable<string> SplitBranchLines(string? chunk)
    {
        var s = chunk ?? string.Empty;
        if (string.IsNullOrWhiteSpace(s))
            yield break;

        // Newlines first (fixes multi-line single callback)
        var parts = s.Split(new[] { '\r', '\n' }, StringSplitOptions.RemoveEmptyEntries);
        foreach (var part in parts)
        {
            var line = part.Trim();
            if (string.IsNullOrWhiteSpace(line))
                continue;
            if (line.StartsWith(">>>", StringComparison.Ordinal))
                continue;

            // Quoted blob leftover: 'origin/A' 'origin/B'
            if ((line.Contains('\'') || line.Contains('"')) && line.Contains(' '))
            {
                foreach (var tok in line.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries))
                {
                    var t = tok.Trim().Trim('\'', '"').Trim();
                    if (!string.IsNullOrWhiteSpace(t))
                        yield return t;
                }
                continue;
            }

            yield return line.Trim().Trim('\'', '"').Trim();
        }
    }

    // origin/GSSv8.7S3 → GSSv8.7S3   (do NOT use Contains("HEAD") on whole chunk)
    private static string? ParseRemoteBranch(string line)
    {
        var t = (line ?? string.Empty).Trim().Trim('\'', '"').Trim();
        if (string.IsNullOrWhiteSpace(t))
            return null;

        if (t.StartsWith("*"))
            t = t.TrimStart('*').Trim().Trim('\'', '"').Trim();

        // symbolic ref line: origin/HEAD -> origin/main
        if (t.Contains("->", StringComparison.Ordinal))
            return null;

        if (t.StartsWith("origin/", StringComparison.OrdinalIgnoreCase))
            t = t.Substring("origin/".Length).Trim().Trim('\'', '"').Trim();

        if (string.IsNullOrWhiteSpace(t))
            return null;
        if (string.Equals(t, "origin", StringComparison.OrdinalIgnoreCase))
            return null;
        if (string.Equals(t, "HEAD", StringComparison.OrdinalIgnoreCase))
            return null;
        if (string.Equals(t, "main", StringComparison.OrdinalIgnoreCase))
            return null; // optional: drop main from DB rollup list

        return t;
    }
