CHANGE 3 — ADD GetBuildFilePath (the combine method)
================================================================================
PASTE inside @code. This is the method that builds the path.

    /// <summary>
    /// Combine Output Root + selected Get Latest From Repo Branch.
    /// Example: F:\build-output + GSSv9.3S1 → F:\build-output\9.3S1
    /// Trims "GSSv" so the folder matches Explorer (9.3S1, 9.2S2).
    /// </summary>
    private string GetBuildFilePath()
    {
        var outputRoot = (OutputRoot ?? string.Empty).Trim().TrimEnd('\\', '/');
        var branch = (RepoFromBranch ?? string.Empty).Trim();

        if (string.IsNullOrWhiteSpace(outputRoot) || string.IsNullOrWhiteSpace(branch))
            return string.Empty;

        // GSSv9.3S1 / origin/GSSv9.3S1 → 9.3S1
        var folder = TrimGssVersionFolder(branch);
        if (string.IsNullOrWhiteSpace(folder))
        {
            var slash = branch.LastIndexOf('/');
            folder = (slash >= 0 && slash < branch.Length - 1)
                ? branch[(slash + 1)..]
                : branch;
        }

        foreach (var c in Path.GetInvalidFileNameChars())
            folder = folder.Replace(c, '_');

        if (string.IsNullOrWhiteSpace(folder))
            return string.Empty;

        return Path.Combine(outputRoot, folder);
    }

CHANGE 4 — ADD BuildFilePathClicked (button handler)
================================================================================
PASTE inside @code:

    private async Task BuildFilePathClicked()
    {
        if (_isBuildingFilePath)
            return;

        _isBuildingFilePath = true;
        try
        {
            var outputRoot = (OutputRoot ?? string.Empty).Trim();
            if (string.IsNullOrWhiteSpace(outputRoot))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Output Root is empty. Browse Output Root first.");
                await InvokeAsync(StateHasChanged);
                return;
            }

            var branch = (RepoFromBranch ?? string.Empty).Trim();
            if (string.IsNullOrWhiteSpace(branch))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Select a branch in Get Latest From Repo Branch (click Refresh first).");
                await InvokeAsync(StateHasChanged);
                return;
            }

            var path = GetBuildFilePath();
            if (string.IsNullOrWhiteSpace(path))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Could not build file path from Output Root + branch.");
                await InvokeAsync(StateHasChanged);
                return;
            }

            Directory.CreateDirectory(path);   // no-op if it already exists
            ResolvedBuildFilePath = path;

            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Build file path: {path}");
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO]   Output Root = {outputRoot}");
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO]   Branch      = {branch}");
            await InvokeAsync(StateHasChanged);

            try
            {
                System.Diagnostics.Process.Start(new System.Diagnostics.ProcessStartInfo
                {
                    FileName = "explorer.exe",
                    Arguments = $"\"{path}\"",
                    UseShellExecute = true
                });
            }
            catch (Exception ex)
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [WARN] Path created but Explorer did not open: {ex.Message}");
            }
        }
        catch (Exception ex)
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Build file path failed: {ex.Message}");
            await InvokeAsync(StateHasChanged);
        }
        finally
        {
            _isBuildingFilePath = false;
            await InvokeAsync(StateHasChanged);
        }
    }

    /// <summary>
    /// Combine Output Root + selected repo branch (trim GSSv).
    /// F:\build-output + GSSv9.3S1 → F:\build-output\9.3S1
    /// </summary>
    public string GetBuildFilePath(string? outputRoot, string? selectedBranch)
    {
        var root = (outputRoot ?? string.Empty).Trim().TrimEnd('\\', '/');
        var branch = (selectedBranch ?? string.Empty).Trim();

        if (string.IsNullOrWhiteSpace(root) || string.IsNullOrWhiteSpace(branch))
            return string.Empty;

        var folder = TrimGssVersionFolder(branch);
        if (string.IsNullOrWhiteSpace(folder))
        {
            var slash = branch.LastIndexOf('/');
            folder = (slash >= 0 && slash < branch.Length - 1)
                ? branch[(slash + 1)..]
                : branch;
        }

        foreach (var c in Path.GetInvalidFileNameChars())
            folder = folder.Replace(c, '_');

        if (string.IsNullOrWhiteSpace(folder))
            return string.Empty;

        return Path.Combine(root, folder);
