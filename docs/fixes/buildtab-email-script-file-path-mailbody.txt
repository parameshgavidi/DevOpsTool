NEW — FULL (list ALL generated script file paths)  ← recommended
--------------------------------------------------------------------------------
Paste this block in place of your existing email try/catch.
Keeps Destination folder AND lists every generated file path.

            // Send success notification email via GSSEmailComponent
            try
            {
                var settings = LoadAppSettings();
                var connectionString = settings.ConnectionString;
                var fromEmail = string.IsNullOrWhiteSpace(settings.FromEmail)
                    ? "GSSHelp@usps.gov"
                    : settings.FromEmail;
                var recipients = settings.EmailRecipients;
                var toEmails = (recipients != null && recipients.Count > 0)
                    ? string.Join(";", recipients)
                    : "devi.s.lam@usps.gov";

                // Collect generated script file full paths (e.g. F:\build-output\db-scripts\combined.sql)
                var scriptPaths = new List<string>();
                if (!string.IsNullOrWhiteSpace(destDir) && Directory.Exists(destDir))
                {
                    scriptPaths = Directory.GetFiles(destDir, "*.*", SearchOption.AllDirectories)
                        .Where(f =>
                        {
                            var ext = Path.GetExtension(f);
                            return ext.Equals(".sql", StringComparison.OrdinalIgnoreCase)
                                   || ext.Equals(".txt", StringComparison.OrdinalIgnoreCase)
                                   || ext.Equals(".prc", StringComparison.OrdinalIgnoreCase);
                        })
                        .OrderBy(f => f, StringComparer.OrdinalIgnoreCase)
                        .ToList();
                }

                var scriptPathsHtml = scriptPaths.Count > 0
                    ? string.Join("<br/>", scriptPaths.Select(p => System.Net.WebUtility.HtmlEncode(p)))
                    : "(no script files found under destination)";

                var mailSubject = $"DB Rollup Bundle Complete - {fromBranch} to {toBranch}";
                var mailBody =
                    $"<html><body>" +
                    $"<p>DB rollup scripts bundled successfully.</p>" +
                    $"<p>From Branch: {System.Net.WebUtility.HtmlEncode(fromBranch)}<br/>" +
                    $"To Branch: {System.Net.WebUtility.HtmlEncode(toBranch)}<br/>" +
                    $"Files Copied: {copied}<br/>" +
                    $"Destination: {System.Net.WebUtility.HtmlEncode(destDir)}<br/>" +
                    $"Generated Script File Location(s):<br/>{scriptPathsHtml}<br/>" +
                    $"Timestamp: {DateTime.Now:yyyy-MM-dd HH:mm:ss}</p>" +
                    $"</body></html>";

                GSS.GSSEmail.SendEMailByNetV2(fromEmail, toEmails, mailSubject, mailBody, connectionString, true);

                LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Email notification sent to {toEmails}\r\n";
            }
            catch (Exception emailEx)
            {
                LogOutput += $"[{DateTime.Now:HH:mm:ss}] [WARN] Failed to send email notification: {emailEx.Message}\r\n";
            }


Expected email body excerpt:

  Generated Script File Location(s):
  F:\build-output\db-scripts\combined.sql
  F:\build-output\db-scripts\combined_Scripts.sql
