ROOT CAUSE (from your BuildTab.razor BuildAll)
----------------------------------------------
Your BuildAll does this:

  1. Set EVERY app Status = "Building"
  2. Task.Run + SemaphoreSlim (up to 4 parallel)
  3. await Task.WhenAll(tasks)

Individual Build runs ONE app on the UI/async path — works.
Build All runs several at once on thread-pool threads → file locks,
shared log/state, Dispatcher issues, WhenAll aggregating failures →
rows end up Failed even though single Build succeeds.

DeployTab InvokeAsync forward is already correct — keep it.

    private async Task BuildAll()
    {
        if (BuildApps.Count == 0)
            return;

        _buildCts?.Cancel();
        _buildCts?.Dispose();
        _buildCts = new CancellationTokenSource();
        var token = _buildCts.Token;

        LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Build All started for {BuildApps.Count} applications\r\n";
        await InvokeAsync(StateHasChanged);

        // Same as clicking Build on each app — ONE AT A TIME
        foreach (var app in BuildApps.ToList())
        {
            if (token.IsCancellationRequested)
                break;

            try
            {
                app.Status = "Building";
                await InvokeAsync(StateHasChanged);

                await BuildApplication(app.Name, token);
            }
            catch (OperationCanceledException)
            {
                app.Status = "Cancelled";
                LogOutput += $"[{DateTime.Now:HH:mm:ss}] [WARN] Build All cancelled\r\n";
                break;
            }
            catch (Exception ex)
            {
                // ONLY this app — never mark every app Failed
                app.Status = "Failed";
                LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] Build All error for {app.Name}: {ex.Message}\r\n";
            }

            await InvokeAsync(StateHasChanged);
        }

        LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Build All finished\r\n";
        await InvokeAsync(StateHasChanged);
    }
