    // 'origin/A' 'origin/B'  →  separate tokens  origin/A , origin/B
    private static IEnumerable<string> SplitBranchTokens(string? line)
    {
        var s = (line ?? string.Empty).Trim();
        if (string.IsNullOrWhiteSpace(s))
            yield break;

        if (s.StartsWith(">>>", StringComparison.Ordinal))
            yield break;

        if ((s.Contains('\'') || s.Contains('"')) && s.Contains(' '))
        {
            foreach (var p in s.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries))
                yield return p;
            yield break;
        }

        yield return s;
    }

    // Strip quotes + origin/  →  GSSv7.7
    private static string NormalizeBranchName(string raw)
    {
        var t = (raw ?? string.Empty).Trim();
        if (string.IsNullOrWhiteSpace(t))
            return string.Empty;

        if (t.StartsWith("*"))
            t = t.TrimStart('*').Trim();

        if (t.Length >= 2 && t[0] == '\'' && t[^1] == '\'')
            t = t.Substring(1, t.Length - 2).Trim();
        if (t.Length >= 2 && t[0] == '"' && t[^1] == '"')
            t = t.Substring(1, t.Length - 2).Trim();
        t = t.Trim().Trim('\'', '"');

        if (string.IsNullOrWhiteSpace(t))
            return string.Empty;
        if (string.Equals(t, "origin", StringComparison.OrdinalIgnoreCase))
            return string.Empty;
        if (t.Contains("HEAD", StringComparison.OrdinalIgnoreCase))
            return string.Empty;
        if (t.StartsWith("origin/", StringComparison.OrdinalIgnoreCase))
            t = t.Substring("origin/".Length);

        return t.Trim();
    }
