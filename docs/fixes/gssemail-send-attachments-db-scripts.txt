SHORT ANSWER
------------
NO — the current SendEmailByNetV2 cannot send attachments as written.

Your method (screenshot):

  public static void SendEmailByNetV2(
      string sFromEmail,
      string sToEmail,
      string sMailSubject,
      string sEmailBody,
      string sGSSConnectionString,
      bool isBodyHtml = false)

- No attachment parameter
- Never calls objMM.Attachments.Add(...)
- MailMessage supports attachments, but this method never uses them

To send DB script files you MUST either:
  A) Extend SendEmailByNetV2 with an optional attachment paths parameter
     (recommended — existing callers keep working), OR
  B) Add a new overload SendEmailByNetV2(..., string[] attachmentPaths)

Earlier BuildTab email guides said "do not edit GSSEmail.cs" for decrypt /
SqlClient fixes. Attachments are different — MailMessage.Attachments only
exists inside GSSEmail, so a small GSSEmail change is required.


WHERE DB SCRIPT FILES COME FROM
-------------------------------
After Bundle Scripts, files are typically under:

  {outputRoot}\db-scripts\          e.g. F:\build-output\db-scripts\
  OR build:dbRollupOutputDirectory  e.g. F:\temp\scripts\

From/To working copies under F:\GSS\DBRollupScript\{branch} are source
checkouts — usually attach the BUNDLED output (db-scripts), not whole repos.

PATH IN MAIL BODY ONLY (no attachments / no GSSEmail.cs change):
  docs/fixes/buildtab-email-script-file-path-mailbody.txt
  Put full generated script path(s) into mailBody, e.g.:
    F:\build-output\db-scripts\combined.sql

CHANGE 2 — BuildTab.razor — attach bundled DB scripts after Bundle
================================================================================
After Bundle Scripts copies files to destDir ({outputRoot}\db-scripts),
collect paths and pass them into SendEmailByNetV2.

Helper (paste in @code):

    private static List<string> CollectDbScriptAttachmentPaths(string folder, int maxFiles = 20)
    {
        var list = new List<string>();
        if (string.IsNullOrWhiteSpace(folder) || !Directory.Exists(folder))
            return list;

        var files = Directory.GetFiles(folder, "*.*", SearchOption.AllDirectories)
            .Where(f =>
            {
                var ext = Path.GetExtension(f);
                return ext.Equals(".sql", StringComparison.OrdinalIgnoreCase)
                       || ext.Equals(".txt", StringComparison.OrdinalIgnoreCase)
                       || ext.Equals(".prc", StringComparison.OrdinalIgnoreCase)
                       || ext.Equals(".zip", StringComparison.OrdinalIgnoreCase);
            })
            .OrderBy(f => f, StringComparer.OrdinalIgnoreCase)
            .Take(maxFiles)
            .ToList();

        list.AddRange(files);
        return list;
    }

In BundleDbScriptsAsync success path (after "Bundle Scripts complete"),
ADD email with attachments (decrypt CS first — same as other email guides):

            // Email bundled DB scripts as attachments
            try
            {
                var settings = LoadAppSettings();

                var fromEmail = string.IsNullOrWhiteSpace(settings.FromEmail)
                    ? "GSSHelp@usps.gov"
                    : settings.FromEmail;

                var recipients = settings.EmailRecipients;
                var toEmails = (recipients != null && recipients.Count > 0)
                    ? string.Join(";", recipients)
                    : "devi.s.lam@usps.gov";

                // Decrypt GSS connection string BEFORE SendEmailByNetV2
                // (see docs/fixes/buildtab-decrypt-before-sendemail.txt)
                var encryptedCs = settings.ConnectionString;
                if (string.IsNullOrWhiteSpace(encryptedCs))
                    throw new InvalidOperationException("GSS connection string missing in appsettings.");

                var crypto = new Encryption.Crypto();
                var plainCs = crypto.DecryptString(encryptedCs); // use DecryptString, not Decrypt

                var attachments = CollectDbScriptAttachmentPaths(destDir);
                if (attachments.Count == 0)
                    AppendLog($"[{DateTime.Now:HH:mm:ss}] [WARN] No DB script files to attach from {destDir}");

                var mailSubject =
                    $"DevOps Tool — DB scripts bundled ({DbRollupFromBranch} → {DbRollupToBranch})";
                var mailBody =
                    $"<html><body>" +
                    $"<p>DB rollup scripts bundled successfully.</p>" +
                    $"<p>From: {DbRollupFromBranch}<br/>To: {DbRollupToBranch}</p>" +
                    $"<p>Output: {destDir}</p>" +
                    $"<p>Attachments: {attachments.Count} file(s)</p>" +
                    $"<p>Timestamp: {DateTime.Now:yyyy-MM-dd HH:mm:ss}</p>" +
                    $"</body></html>";

                GSS.GSSEmail.SendEmailByNetV2(
                    fromEmail,
                    toEmails,
                    mailSubject,
                    mailBody,
                    plainCs,
                    true,
                    attachments);   // NEW — attachment paths

                AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Email with {attachments.Count} attachment(s) sent to {toEmails}");
            }
            catch (Exception emailEx)
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [WARN] Email with attachments failed: {emailEx.Message}");
            }

CHANGE 3 — Optional: zip first if many / large scripts
================================================================================
SMTP size limits often fail with many .sql files. Prefer one zip:

    private static string? ZipDbScriptsFolder(string folder, string zipPath)
    {
        if (!Directory.Exists(folder))
            return null;
        if (File.Exists(zipPath))
            File.Delete(zipPath);
        System.IO.Compression.ZipFile.CreateFromDirectory(folder, zipPath);
        return File.Exists(zipPath) ? zipPath : null;
    }

Then:

    var zip = Path.Combine(Path.GetTempPath(),
        $"db-scripts-{DbRollupFromBranch}-to-{DbRollupToBranch}-{DateTime.Now:yyyyMMddHHmmss}.zip");
