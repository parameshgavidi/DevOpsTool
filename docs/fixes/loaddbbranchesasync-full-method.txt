    private async Task LoadDbBranchesAsync()
    {
        if (_isLoadingBranches)
            return;

        _isLoadingBranches = true;
        try
        {
            // Use RepoRoot (gss-root), NOT Script Source (F:\db-rollup-scripts)
            var root = RepoRoot?.Trim() ?? string.Empty;
            if (string.IsNullOrWhiteSpace(root) || !Directory.Exists(root))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Repo Root not found: {root}. Set Repo Root to your Git repo (gss-root).");
                await InvokeAsync(StateHasChanged);
                return;
            }

            var settings = LoadAppSettings();
            var command = settings?.Build?.DbRollupPullBranchesCommand?.Trim();
            if (string.IsNullOrWhiteSpace(command))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] build:dbRollupPullBranchesCommand missing in appsettings.json.");
                await InvokeAsync(StateHasChanged);
                return;
            }

            var lines = new List<string>();

            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Loading DB branches from {root}...");
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Command: {command}");
            await InvokeAsync(StateHasChanged);

            var exit = await BuildService.RunCommandAsync(
                root,
                command,
                line =>
                {
                    foreach (var raw in SplitBranchTokens(line))
                    {
                        var t = NormalizeBranchName(raw);
                        if (string.IsNullOrWhiteSpace(t))
                            continue;
                        if (!t.StartsWith("GSS", StringComparison.OrdinalIgnoreCase))
                            continue;
                        if (!lines.Contains(t, StringComparer.OrdinalIgnoreCase))
                            lines.Add(t);
                    }
                },
                CancellationToken.None);

            if (exit != 0)
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Failed to list branches (exit {exit}).");
                await InvokeAsync(StateHasChanged);
                return;
            }

            lines.Sort(StringComparer.OrdinalIgnoreCase);
            DbBranchOptions = lines;

            if (DbBranchOptions.Count == 0)
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Parsed 0 branches. Check command output / Repo Root.");
            }
            else if (DbBranchOptions.Count == 1 && DbBranchOptions[0].Contains('\''))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Still one quoted blob — old parser still running. Delete old LoadDbBranchesAsync and paste again.");
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] blob={DbBranchOptions[0]}");
            }
            else
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Loaded {DbBranchOptions.Count} branch(es) into dropdown.");
                var preview = Math.Min(8, DbBranchOptions.Count);
                for (var i = 0; i < preview; i++)
                    AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO]   [{i}] {DbBranchOptions[i]}");
            }

            if (DbBranchOptions.Count > 0)
            {
                if (string.IsNullOrWhiteSpace(DbRollupFromBranch)
                    || !DbBranchOptions.Contains(DbRollupFromBranch, StringComparer.OrdinalIgnoreCase)
                    || DbRollupFromBranch.Contains('\''))
                {
                    DbRollupFromBranch = DbBranchOptions[0];
                }

                if (string.IsNullOrWhiteSpace(DbRollupToBranch)
                    || !DbBranchOptions.Contains(DbRollupToBranch, StringComparer.OrdinalIgnoreCase)
                    || DbRollupToBranch.Contains('\''))
                {
                    DbRollupToBranch = DbBranchOptions.Count > 1
                        ? DbBranchOptions[1]
                        : DbBranchOptions[0];
                }
            }

            await InvokeAsync(StateHasChanged);
        }
        catch (Exception ex)
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Load branches failed: {ex.Message}");
            await InvokeAsync(StateHasChanged);
        }
        finally
        {
            _isLoadingBranches = false;
            await InvokeAsync(StateHasChanged);
        }
    }

    private static IEnumerable<string> SplitBranchTokens(string? line)
    {
        var s = (line ?? string.Empty).Trim();
        if (string.IsNullOrWhiteSpace(s))
            yield break;
        if (s.StartsWith(">>>", StringComparison.Ordinal))
            yield break;

        // Quoted blob: 'origin/A' 'origin/B'  OR mangled: GSS8.7' 'origin/B'
        if (s.Contains('\'') || s.Contains('"'))
        {
            foreach (var p in s.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries))
                yield return p;
            yield break;
        }

        yield return s;
    }

    private static string NormalizeBranchName(string raw)
    {
        var t = (raw ?? string.Empty).Trim();
        if (string.IsNullOrWhiteSpace(t))
            return string.Empty;

        if (t.StartsWith("*"))
            t = t.TrimStart('*').Trim();

        t = t.Trim().Trim('\'', '"').Trim();

        if (string.IsNullOrWhiteSpace(t))
            return string.Empty;
        if (string.Equals(t, "origin", StringComparison.OrdinalIgnoreCase))
            return string.Empty;
        if (t.Contains("HEAD", StringComparison.OrdinalIgnoreCase))
            return string.Empty;

        while (t.StartsWith("origin/", StringComparison.OrdinalIgnoreCase))
            t = t.Substring("origin/".Length).Trim().Trim('\'', '"').Trim();

        return t.Trim();
    }

    protected override async Task OnAfterRenderAsync(bool firstRender)
    {
        if (!firstRender)
            return;

        if (DbBranchOptions.Count == 0
            && !string.IsNullOrWhiteSpace(RepoRoot)
            && Directory.Exists(RepoRoot))
        {
            await LoadDbBranchesAsync();
        }
    }
