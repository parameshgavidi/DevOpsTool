  "name": "GSS API",
  "type": "DotNetCoreApi",
  "projectPath": "Web Services\\GSSAPI\\GSSAPI\\GSSAPI.csproj",
  "gitLatestCommand": "git stash push -m \"devops-pre-build\" -- \"Web Services\\GSSAPI\" && git fetch && git restore --source=@{u} --staged --worktree -- \"Web Services\\GSSAPI\" && git stash pop",
  "buildCommand": "dotnet publish \"{projectFile}\" -c Release -o {outputSubfolder}",
  "outputSubfolder": "GSSAPI"

gitLatestCommand:

  git stash push -m "devops-pre-build" -- "Web Services\GSSAPI" && git fetch && git restore --source=@{u} --staged --worktree -- "Web Services\GSSAPI" && git stash pop

buildCommand:

  dotnet publish "{projectFile}" -c Release -o {outputSubfolder}


--------------------------------------------------------------------------------
Notes
--------------------------------------------------------------------------------
- cwd = Code Repo Root (build:repoRoot) — same as GSS Web Service / GSSWeb.
- gitLatest path is the GSSAPI folder under Web Services (not only the .csproj).
- Pattern matches other apps: stash → fetch → restore from @{u} → stash pop
  (keeps local changes; does NOT use git clean -fd).
- {projectFile} = resolved projectPath (GSSAPI.csproj).
- {outputSubfolder} = GSSAPI (publish output folder name).
- Get latest is toolbar-only; Build does not run gitLatestCommand.
  See: docs/fixes/buildtab-no-getlatest-on-build.txt
- DotNetCoreApi uses dotnet publish (not msbuild / PrecompiledWeb).
