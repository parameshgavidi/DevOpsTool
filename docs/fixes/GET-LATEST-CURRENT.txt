E) LoadApplicationsFromConfigAsync — ONE method

    private Task LoadApplicationsFromConfigAsync()
    {
        var settings = LoadAppSettings();
        var loadedApps = settings?.Build?.Applications;

        if (loadedApps is { Count: > 0 })
            BuildApps = loadedApps;
        else
            BuildApps = new List<Application>();

        foreach (var app in BuildApps)
        {
            if (string.IsNullOrWhiteSpace(app.Status))
                app.Status = "Ready";
        }

        return Task.CompletedTask;
    }

--------------------------------------------------------------------------------
Used by Get latest button AFTER branch checkout. NOT used by Build.

    private async Task<bool> GetLatestFromRepoAsync(Application app, string workingDir, CancellationToken cancellation)
    {
        if (string.IsNullOrWhiteSpace(app.GitLatestCommand))
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] No gitLatestCommand for {app.Name} — skipping");
            return true;
        }

        var gitCmd = app.GitLatestCommand.Trim();
        var gitWorkingDir = string.IsNullOrWhiteSpace(RepoRoot) ? workingDir : RepoRoot.Trim();

        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Getting latest for {app.Name}...");
        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Git: {gitCmd}  (cwd: {gitWorkingDir})");
        await InvokeAsync(StateHasChanged);

        var gitExit = await BuildService.RunCommandAsync(
            gitWorkingDir,
            gitCmd,
            line =>
            {
                lock (_pendingLogBuffer)
                {
                    _pendingLogBuffer.Append(line);
                    if (!line.EndsWith('\n') && !line.EndsWith('\r'))
                        _pendingLogBuffer.AppendLine();
                }
            },
            cancellation);

        FlushPendingLogs();

        if (gitExit != 0)
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Git latest failed for {app.Name} with exit code {gitExit}");
            await InvokeAsync(StateHasChanged);
            return false;
        }

        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Git latest OK for {app.Name}");
        await InvokeAsync(StateHasChanged);
        return true;
    }
