E) LoadApplicationsFromConfigAsync — ONE method
--------------------------------------------------------------------------------

    private Task LoadApplicationsFromConfigAsync()
    {
        var settings = LoadAppSettings();
        var loadedApps = settings?.Build?.Applications;

        if (loadedApps is { Count: > 0 })
            BuildApps = loadedApps;
        else
            BuildApps = new List<Application>();

        foreach (var app in BuildApps)
        {
            if (string.IsNullOrWhiteSpace(app.Status))
                app.Status = "Ready";
        }

        return Task.CompletedTask;
    }

F) GetLatestFromRepoAsync — KEEP THIS (reads appsettings gitLatestCommand)

Used by Get latest button. NOT used by Build.

    private async Task<bool> GetLatestFromRepoAsync(Application app, string workingDir, CancellationToken cancellation)
    {
        if (string.IsNullOrWhiteSpace(app.GitLatestCommand))
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] No gitLatestCommand for {app.Name} — skipping");
            return true;
        }

        var gitCmd = app.GitLatestCommand.Trim();
        var gitWorkingDir = string.IsNullOrWhiteSpace(RepoRoot) ? workingDir : RepoRoot.Trim();

        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Getting latest for {app.Name}...");
        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Git: {gitCmd}  (cwd: {gitWorkingDir})");
        await InvokeAsync(StateHasChanged);

        var gitExit = await BuildService.RunCommandAsync(
            gitWorkingDir,
            gitCmd,
            line =>
            {
                lock (_pendingLogBuffer)
                {
                    _pendingLogBuffer.Append(line);
                    if (!line.EndsWith('\n') && !line.EndsWith('\r'))
                        _pendingLogBuffer.AppendLine();
                }
            },
            cancellation);

        FlushPendingLogs();

        if (gitExit != 0)
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Git latest failed for {app.Name} with exit code {gitExit}");
            await InvokeAsync(StateHasChanged);
            return false;
        }

        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Git latest OK for {app.Name}");
        await InvokeAsync(StateHasChanged);
        return true;
    }

G) GetLatestAndShowAppsAsync — REPLACE your old method with THIS
--------------------------------------------------------------------------------
NO hardcoded "git fetch && git pull --ff-only".
Loads all apps, then calls GetLatestFromRepoAsync for each.

    private async Task GetLatestAndShowAppsAsync()
    {
        if (_isGettingLatest)
            return;

        _isGettingLatest = true;
        _showApplications = false;
        try
        {
            var root = RepoRoot?.Trim() ?? string.Empty;
            if (string.IsNullOrWhiteSpace(root) || !Directory.Exists(root))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Repo Root not found: {root}");
                await InvokeAsync(StateHasChanged);
                return;
            }

            var gitDir = Path.Combine(root, ".git");
            if (!Directory.Exists(gitDir) && !File.Exists(gitDir))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Not a git repo (missing .git): {root}");
                await InvokeAsync(StateHasChanged);
                return;
            }

            // 1) Load ALL apps from appsettings
            await LoadApplicationsFromConfigAsync();

            if (BuildApps.Count == 0)
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] No applications in appsettings (build:applications)");
                await InvokeAsync(StateHasChanged);
                return;
            }

            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Get latest for {BuildApps.Count} app(s) using gitLatestCommand from appsettings...");
            await InvokeAsync(StateHasChanged);

            StartLogFlushTimer();

            var okCount = 0;
            var failCount = 0;

            // 2) Call GetLatestFromRepoAsync for EACH app
            foreach (var app in BuildApps)
            {
                app.Status = "Getting latest...";
                await InvokeAsync(StateHasChanged);

                var ok = await GetLatestFromRepoAsync(app, root, CancellationToken.None);
                if (ok)
                {
                    app.Status = "Ready";
                    okCount++;
                }
                else
                {
                    app.Status = "Failed";
                    failCount++;
                }

                await InvokeAsync(StateHasChanged);
            }

            FlushPendingLogs();

            // 3) Show grid
            _showApplications = true;
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Get latest complete — OK: {okCount}, Failed: {failCount}");
            await InvokeAsync(StateHasChanged);
        }
        catch (Exception ex)
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Get latest failed: {ex.Message}");
            _showApplications = false;
            await InvokeAsync(StateHasChanged);
        }
        finally
        {
            StopLogFlushTimer();
            _isGettingLatest = false;
            await InvokeAsync(StateHasChanged);
        }
    }
