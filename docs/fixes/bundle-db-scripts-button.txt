================================================================================
Bundle Scripts button — call DBRollupScriptBuilder.exe and copy to output
Settings come from appsettings.json (do NOT hardcode paths in C#)
================================================================================

FLOW
----
1. User clicks "Bundle Scripts"
2. Run DBRollupScriptBuilder.exe (path from appsettings)
3. Exe writes scripts to its OutputDirectory (path from appsettings)
4. Copy script files to {outputRoot}\db-scripts\
5. Stream stdout/stderr to OUTPUT LOG
6. On error → [ERROR] in OUTPUT LOG


PART A — appsettings.json (DevOps tool)
--------------------------------------------------------------------------------
Put all paths under "build" (same section as outputRoot / dbScriptSource):

{
  "build": {
    "repoRoot": "F:\\GSS\\AppDev\\gss-root",
    "outputRoot": "F:\\build-output",
    "dbScriptSource": "F:\\db-rollup-scripts",
    "dbRollupExePath": "F:\\GSS\\AppDev\\gss-utilities\\DBRollupScriptBuilder\\DBRollupScriptBuilder\\bin\\Debug\\net10.0\\DBRollupScriptBuilder.exe",
    "dbRollupOutputDirectory": "F:\\temp\\scripts",
    "apps": [ ]
  }
}

Mapping:
  build:outputRoot               -> OutputRoot
  build:dbScriptSource           -> DbScriptSource  (Script Source textbox)
  build:dbRollupExePath          -> DbRollupExePath
  build:dbRollupOutputDirectory  -> DbRollupOutputDirectory

dbRollupOutputDirectory MUST match the exe's own appsettings
ApplicationSettings:OutputDirectory (f:\temp\scripts).


PART B — Model properties (DeploymentModels / BuildSettings)
--------------------------------------------------------------------------------
Add to the same class that already has OutputRoot / DbScriptSource:

    public string DbRollupExePath { get; set; } = string.Empty;
    public string DbRollupOutputDirectory { get; set; } = string.Empty;

No hardcoded path defaults in C# — empty until loaded from config.


PART C — Load from IConfiguration in BuildTab.razor
--------------------------------------------------------------------------------
Wherever you already load OutputRoot from appsettings (OnInitialized /
LoadBuildSettings), also load the new fields:

    // Existing:
    // OutputRoot = buildSettings.OutputRoot;
    // DbScriptSource = buildSettings.DbScriptSource;

    DbRollupExePath = buildSettings.DbRollupExePath ?? string.Empty;
    DbRollupOutputDirectory = buildSettings.DbRollupOutputDirectory ?? string.Empty;

Or with IConfiguration:

    DbRollupExePath = Configuration["build:dbRollupExePath"] ?? string.Empty;
    DbRollupOutputDirectory = Configuration["build:dbRollupOutputDirectory"] ?? string.Empty;

Fields in BuildTab.razor @code (no path literals):

    private bool _isBundling;
    private string DbRollupExePath { get; set; } = string.Empty;
    private string DbRollupOutputDirectory { get; set; } = string.Empty;
    // OutputRoot, DbScriptSource — already exist; keep loading them from appsettings

_isBundling:
  ENABLE  → _isBundling = true;   at start of BundleDbScriptsAsync
  DISABLE → _isBundling = false;  in finally

Button:
  <button disabled="@_isBundling" @onclick="BundleDbScriptsAsync">Bundle Scripts</button>


PART D — BundleDbScriptsAsync (uses settings properties only)
--------------------------------------------------------------------------------

    private async Task BundleDbScriptsAsync()
    {
        if (_isBundling)
            return;

        _isBundling = true;
        try
        {
            var exePath = DbRollupExePath?.Trim();
            if (string.IsNullOrWhiteSpace(exePath) || !File.Exists(exePath))
            {
                LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] DBRollupScriptBuilder.exe not found: {exePath}. Check build:dbRollupExePath in appsettings.json.\r\n";
                await InvokeAsync(StateHasChanged);
                return;
            }

            var exeDir = Path.GetDirectoryName(exePath)!;
            var builderOutputDir = DbRollupOutputDirectory?.Trim() ?? string.Empty;
            if (string.IsNullOrWhiteSpace(builderOutputDir))
            {
                LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] build:dbRollupOutputDirectory is missing in appsettings.json.\r\n";
                await InvokeAsync(StateHasChanged);
                return;
            }

            if (string.IsNullOrWhiteSpace(OutputRoot))
            {
                LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] build:outputRoot is missing in appsettings.json.\r\n";
                await InvokeAsync(StateHasChanged);
                return;
            }

            var destDir = Path.Combine(OutputRoot.TrimEnd('\\', '/'), "db-scripts");

            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Bundling DB scripts...\r\n";
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Exe: {exePath}\r\n";
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Script source: {DbScriptSource}\r\n";
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Builder output: {builderOutputDir}\r\n";
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Copy destination: {destDir}\r\n";
            await InvokeAsync(StateHasChanged);

            var args = string.Empty; // add -f / -t here later if needed

            StartLogFlushTimer();
            var exit = await BuildService.RunCommandAsync(
                exeDir,
                string.IsNullOrEmpty(args) ? $"\"{exePath}\"" : $"\"{exePath}\" {args}",
                line =>
                {
                    lock (_pendingLogBuffer)
                    {
                        _pendingLogBuffer.Append(line);
                        if (!line.EndsWith('\n') && !line.EndsWith('\r'))
                            _pendingLogBuffer.AppendLine();
                    }
                });

            FlushPendingLogs();

            if (exit != 0)
            {
                LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] DBRollupScriptBuilder failed with exit code {exit}\r\n";
                await InvokeAsync(StateHasChanged);
                return;
            }

            if (!Directory.Exists(builderOutputDir))
            {
                LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] Builder output folder not found: {builderOutputDir}\r\n";
                await InvokeAsync(StateHasChanged);
                return;
            }

            var scriptFiles = Directory.GetFiles(builderOutputDir, "*.*", SearchOption.AllDirectories)
                .Where(f =>
                {
                    var ext = Path.GetExtension(f);
                    return ext.Equals(".sql", StringComparison.OrdinalIgnoreCase)
                           || ext.Equals(".txt", StringComparison.OrdinalIgnoreCase)
                           || ext.Equals(".prc", StringComparison.OrdinalIgnoreCase);
                })
                .ToList();

            if (scriptFiles.Count == 0)
            {
                LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] No script files found in {builderOutputDir}\r\n";
                await InvokeAsync(StateHasChanged);
                return;
            }

            Directory.CreateDirectory(destDir);
            var copied = 0;
            foreach (var file in scriptFiles)
            {
                var relative = Path.GetRelativePath(builderOutputDir, file);
                var dest = Path.Combine(destDir, relative);
                Directory.CreateDirectory(Path.GetDirectoryName(dest)!);
                File.Copy(file, dest, overwrite: true);
                copied++;
                LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Copied {relative}\r\n";
            }

            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Bundle Scripts complete — {copied} file(s) copied to {destDir}\r\n";
            await InvokeAsync(StateHasChanged);
        }
        catch (Exception ex)
        {
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] Bundle Scripts failed: {ex.Message}\r\n";
            await InvokeAsync(StateHasChanged);
        }
        finally
        {
            StopLogFlushTimer();
            _isBundling = false;
            await InvokeAsync(StateHasChanged);
        }
    }


PART E — checklist
--------------------------------------------------------------------------------
[ ] Paths only in appsettings.json (not hardcoded in BuildTab.razor)
[ ] Model has DbRollupExePath + DbRollupOutputDirectory
[ ] BuildTab loads them with OutputRoot / DbScriptSource
[ ] Bundle Scripts uses those properties
[ ] finally sets _isBundling = false
[ ] On success: scripts under {outputRoot}\db-scripts\
