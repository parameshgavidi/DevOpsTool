    private async Task LoadDbScriptFoldersAsync()
    {
        if (_isLoadingBranches)
            return;

        _isLoadingBranches = true;
        try
        {
            var root = ScriptSource?.Trim() ?? string.Empty;
            if (string.IsNullOrWhiteSpace(root) || !Directory.Exists(root))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Script Source not found: {root}. Expected ...\\gss-database\\Scripts");
                await InvokeAsync(StateHasChanged);
                return;
            }

            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Loading DB script folders from {root}...");
            await InvokeAsync(StateHasChanged);

            var folders = Directory.GetDirectories(root)
                .Select(Path.GetFileName)
                .Where(n => !string.IsNullOrWhiteSpace(n))
                .Cast<string>()
                .Where(n => !n.StartsWith(".", StringComparison.Ordinal))
                .ToList();

            folders.Sort(CompareScriptFolderNames);
            DbBranchOptions = folders;

            if (DbBranchOptions.Count == 0)
            {
                DbRollupToBranch = string.Empty;
                DbRollupFromBranch = string.Empty;
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] No folders under Script Source.");
            }
            else
            {
                // Default To = latest version folder (skip Prod for default if present)
                var latest = PickLatestScriptFolder(DbBranchOptions);
                DbRollupToBranch = latest;
                DbRollupFromBranch = GetPreviousScriptFolder(DbBranchOptions, latest) ?? string.Empty;

                AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Loaded {DbBranchOptions.Count} folder(s). To={DbRollupToBranch}, From={DbRollupFromBranch}");
                var preview = Math.Min(8, DbBranchOptions.Count);
                for (var i = 0; i < preview; i++)
                    AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO]   [{i}] {DbBranchOptions[i]}");
            }

            await InvokeAsync(StateHasChanged);
        }
        catch (Exception ex)
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Load script folders failed: {ex.Message}");
            await InvokeAsync(StateHasChanged);
        }
        finally
        {
            _isLoadingBranches = false;
            await InvokeAsync(StateHasChanged);
        }
    }

    private Task OnToBranchChanged(ChangeEventArgs e)
    {
        var to = e.Value?.ToString()?.Trim() ?? string.Empty;
        DbRollupToBranch = to;
        DbRollupFromBranch = GetPreviousScriptFolder(DbBranchOptions, to) ?? string.Empty;
        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] To Branch={DbRollupToBranch} → From Branch={DbRollupFromBranch}");
        return InvokeAsync(StateHasChanged);
    }

    /// <summary>
    /// Version-aware sort: 8.9 &lt; 8.21 &lt; 9.0 &lt; 9.2; non-numeric (Prod) last.
    /// </summary>
    private static int CompareScriptFolderNames(string a, string b)
    {
        var va = TryParseVersionParts(a);
        var vb = TryParseVersionParts(b);
        if (va is not null && vb is not null)
        {
            var n = Math.Max(va.Count, vb.Count);
            for (var i = 0; i < n; i++)
            {
                var x = i < va.Count ? va[i] : 0;
                var y = i < vb.Count ? vb[i] : 0;
                if (x != y) return x.CompareTo(y);
            }
            return 0;
        }
        if (va is not null) return -1; // versions before Prod/text
        if (vb is not null) return 1;
        return string.Compare(a, b, StringComparison.OrdinalIgnoreCase);
    }

    private static List<int>? TryParseVersionParts(string name)
    {
        if (string.IsNullOrWhiteSpace(name)) return null;
        var parts = name.Split('.', StringSplitOptions.RemoveEmptyEntries);
        if (parts.Length == 0) return null;
        var list = new List<int>(parts.Length);
        foreach (var p in parts)
        {
            if (!int.TryParse(p, out var n))
                return null;
            list.Add(n);
        }
        return list;
    }

    private static string PickLatestScriptFolder(List<string> options)
    {
        // Prefer highest numeric version; ignore trailing Prod for default To
        for (var i = options.Count - 1; i >= 0; i--)
        {
            if (TryParseVersionParts(options[i]) is not null)
                return options[i];
        }
        return options[^1];
    }

    private static string? GetPreviousScriptFolder(List<string> options, string toBranch)
    {
        if (options.Count == 0 || string.IsNullOrWhiteSpace(toBranch))
            return null;

        var idx = options.FindIndex(x =>
            string.Equals(x, toBranch, StringComparison.OrdinalIgnoreCase));
        if (idx <= 0)
            return null; // first item has no previous

        return options[idx - 1];
    }

    private async Task<bool> GetLatestDbProjectAsync()
    {
        var settings = LoadAppSettings();
        var scriptSource = ScriptSource?.Trim();
        if (string.IsNullOrWhiteSpace(scriptSource))
            scriptSource = settings?.Build?.DbScriptSource?.Trim();

        if (string.IsNullOrWhiteSpace(scriptSource))
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] dbScriptSource / Script Source is empty — cannot get latest DB project.");
            return false;
        }

        // Git repo root is parent of Scripts (…\gss-database), or ScriptSource itself if it has .git
        var gitRoot = ResolveDbGitRoot(scriptSource);
        if (string.IsNullOrWhiteSpace(gitRoot) || !Directory.Exists(gitRoot))
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] DB git root not found from Script Source: {scriptSource}");
            return false;
        }

        var gitCmd = settings?.Build?.GitLatestCommand?.Trim();
        if (string.IsNullOrWhiteSpace(gitCmd))
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] build:gitLatestCommand missing in appsettings.json (build-level, not under apps).");
            return false;
        }

        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Getting latest DB project...");
        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] DB Script Source: {scriptSource}");
        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] DB Git root: {gitRoot}");
        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Git: {gitCmd}");
        await InvokeAsync(StateHasChanged);

        var exit = await BuildService.RunCommandAsync(
            gitRoot,
            gitCmd,
            line =>
            {
                lock (_pendingLogBuffer)
                {
                    _pendingLogBuffer.Append(line);
                    if (!line.EndsWith('\n') && !line.EndsWith('\r'))
                        _pendingLogBuffer.AppendLine();
                }
            },
            CancellationToken.None);

        FlushPendingLogs();

        if (exit != 0)
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] DB get latest failed (exit {exit})");
            await InvokeAsync(StateHasChanged);
            return false;
        }

        AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] DB get latest OK");

        // Refresh To/From folder list after pull (new folders may appear)
        await LoadDbScriptFoldersAsync();
        await InvokeAsync(StateHasChanged);
        return true;
    }

    private static string ResolveDbGitRoot(string scriptSource)
    {
        var path = scriptSource.Trim().TrimEnd('\\', '/');
        if (Directory.Exists(Path.Combine(path, ".git")) || File.Exists(Path.Combine(path, ".git")))
            return path;

        // …\gss-database\Scripts → …\gss-database
        var parent = Directory.GetParent(path)?.FullName;
        if (!string.IsNullOrWhiteSpace(parent)
            && (Directory.Exists(Path.Combine(parent, ".git")) || File.Exists(Path.Combine(parent, ".git"))))
            return parent;

        // Fall back to parent even if .git check fails (bare / unusual setups)
        return parent ?? path;
    }
