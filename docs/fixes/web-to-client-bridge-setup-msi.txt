--------------------------------------------------------------------------------
ROOT CAUSE — cmd.exe /c quoting (same bug as Bundle Scripts)
--------------------------------------------------------------------------------
BuildService runs:  cmd.exe /c {command}

This form BREAKS when the exe path has spaces (Program Files):

  "F:\Program Files\...\devenv.com" "....sln" /build Release ...

cmd treats that as a bad invocation → devenv shows help → exit 1 → no MSI.

FIX (leading extra quote — required by cmd.exe /c), same pattern as
docs/fixes/bundle-db-scripts-button.txt:

  ""F:\Program Files\...\devenv.com" "....sln" /build Release /project "Web2ClientBridge.Setup" /projectconfig Release"

In C#:
  var command = $"\"\"{devenvPath}\" \"{setupSolutionPath}\" /build Release /project \"{setupProjectName}\" /projectconfig Release\"";

Do NOT put setupBuildCommand in appsettings as a pre-quoted string that
omits the leading "". Build the command in BuildSetupProjectAsync in code.

PART D — BuildSetupProjectAsync (REPLACE — cmd-safe devenv + copy MSI to output)
--------------------------------------------------------------------------------
Critical changes vs your current method:
  1) Build devenv command with leading "" for cmd.exe /c
  2) Prefer devenv.com when present (console output); else devenv.exe

    private async Task<int> BuildSetupProjectAsync(
        Application app, string outputFolder, CancellationToken cancellation)
    {
        var setupProjectFile = GetConfiguredPath(app.SetupProjectPath);
        if (string.IsNullOrWhiteSpace(setupProjectFile) || !File.Exists(setupProjectFile))
        {
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] Setup project not found for {app.Name}: {app.SetupProjectPath}\r\n";
            return -1;
        }

        var setupProjectDir = Path.GetDirectoryName(setupProjectFile)!;
        var setupProjectName = Path.GetFileNameWithoutExtension(setupProjectFile);

        string? setupSolutionPath = null;
        if (!string.IsNullOrWhiteSpace(app.SetupSolutionPath))
            setupSolutionPath = GetConfiguredPath(app.SetupSolutionPath);

        if (string.IsNullOrWhiteSpace(setupSolutionPath) || !File.Exists(setupSolutionPath))
        {
            var parent = Path.GetDirectoryName(setupProjectDir);
            var candidate = parent is null ? null : Path.Combine(parent, "WebToClientBridge.sln");
            if (!string.IsNullOrEmpty(candidate) && File.Exists(candidate))
                setupSolutionPath = candidate;
            else if (!string.IsNullOrEmpty(parent))
                setupSolutionPath = Directory.GetFiles(parent, "*.sln").FirstOrDefault();
        }

        if (string.IsNullOrWhiteSpace(setupSolutionPath) || !File.Exists(setupSolutionPath))
        {
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] Setup solution (.sln) not found for {app.Name}\r\n";
            return -1;
        }

        var devenvPath = BuildService.ResolveVisualStudioIdePath();
        if (string.IsNullOrWhiteSpace(devenvPath) || !File.Exists(devenvPath))
        {
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] devenv not found. Install Visual Studio + Installer Projects for .vdproj.\r\n";
            return -1;
        }

        // Prefer .com for console stdout (your log already resolved devenv.com)
        var devenvCom = Path.ChangeExtension(devenvPath, ".com");
        if (!string.IsNullOrEmpty(devenvCom) && File.Exists(devenvCom))
            devenvPath = devenvCom;

        // IMPORTANT: BuildService wraps with cmd.exe /c {command}
        // Same fix as Bundle Scripts — leading extra quote required.
        string command;
        if (!string.IsNullOrWhiteSpace(app.SetupBuildCommand))
        {
            command = app.SetupBuildCommand
                .Replace("{setupProjectFile}", setupProjectFile, StringComparison.OrdinalIgnoreCase)
                .Replace("{setupProjectDir}", setupProjectDir, StringComparison.OrdinalIgnoreCase)
                .Replace("{setupSolutionPath}", setupSolutionPath, StringComparison.OrdinalIgnoreCase)
                .Replace("{setupProjectName}", setupProjectName, StringComparison.OrdinalIgnoreCase)
                .Replace("{devenvPath}", devenvPath, StringComparison.OrdinalIgnoreCase)
                .Replace("{outputFolder}", outputFolder, StringComparison.OrdinalIgnoreCase);

            // If appsettings used the broken single-quoted form, upgrade it.
            if (command.StartsWith("\"", StringComparison.Ordinal)
                && !command.StartsWith("\"\"", StringComparison.Ordinal))
            {
                command = "\"" + command + "\"";
            }
        }
        else
        {
            command =
                $"\"\"{devenvPath}\" \"{setupSolutionPath}\" /build Release /project \"{setupProjectName}\" /projectconfig Release\"";
        }

        var solutionDir = Path.GetDirectoryName(setupSolutionPath) ?? setupProjectDir;

        LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Building setup MSI for {app.Name}...\r\n";
        LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Setup command: {command}\r\n";

        StartLogFlushTimer();
        var exit = await BuildService.RunCommandAsync(
            solutionDir,
            command,
            line =>
            {
                lock (_pendingLogBuffer)
                {
                    _pendingLogBuffer.Append(line);
                }
                app.LogEntries.Add(line.TrimEnd('\r', '\n'));
            },
            cancellation);
        FlushPendingLogs();

        if (exit != 0)
        {
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] Setup build failed for {app.Name} with exit code {exit}\r\n";
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] If log shows devenv HELP (/useenv, /TfsLink), cmd quoting is still wrong — command must start with \"\".\r\n";
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] If quoting looks correct, install VS 'Installer Projects' and rebuild Web2ClientBridge.Setup once in the IDE.\r\n";
            return exit;
        }

        // Find newest *.msi under the setup project (Release / Debug / nested)
        var msiFiles = Directory.Exists(setupProjectDir)
            ? Directory.GetFiles(setupProjectDir, "*.msi", SearchOption.AllDirectories)
                .OrderByDescending(File.GetLastWriteTimeUtc)
                .ToArray()
            : Array.Empty<string>();

        if (msiFiles.Length == 0)
        {
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [ERROR] Setup MSI not found under {setupProjectDir} after devenv build.\r\n";
            return -1;
        }

        Directory.CreateDirectory(outputFolder);

        // Copy every MSI into F:\build-output\Web2ClientBridge\  (folder you open in Explorer)
        foreach (var msi in msiFiles)
        {
            var dest = Path.Combine(outputFolder, Path.GetFileName(msi));
            File.Copy(msi, dest, overwrite: true);
            LogOutput += $"[{DateTime.Now:HH:mm:ss}] [INFO] Setup MSI copied to {dest}\r\n";
        }

        return 0;
    }
