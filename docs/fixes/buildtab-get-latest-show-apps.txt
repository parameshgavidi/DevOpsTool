BUILD tab — Get latest button, then show APPLICATIONS grid
================================================================================

PROBLEM
-------
BUILD CONFIGURATION shows Repo Root / Output Root, but there is no Get latest
button. The APPLICATIONS grid is empty (headers only) and should appear AFTER
Get latest succeeds — matching the BUILD flow:

  Configure paths → Get latest (pull code) → Display applications → Build


FLOW
----
1. User sets Repo Root / Output Root (Browse or appsettings)
2. Click Get latest
3. Validate Repo Root is a git working tree
4. git fetch + git pull --ff-only at Code Repo Root
5. Load apps from build:apps (appsettings) into the Applications list
6. Show the APPLICATIONS grid (hidden until this step succeeds)
7. User can then Build / Build All / Bundle Scripts

4) BuildTab.razor — REPLACE/ADD GetLatestAndShowAppsAsync
--------------------------------------------------------------------------------
This is the BUILD-tab toolbar action (full-repo pull). It is separate from
per-app GetLatestFromRepoAsync used inside BuildApplication
(see docs/fixes/git-latest-before-build.txt).

    private async Task GetLatestAndShowAppsAsync()
    {
        if (_isGettingLatest)
            return;

        _isGettingLatest = true;
        _showApplications = false;
        try
        {
            var root = RepoRoot?.Trim() ?? string.Empty;
            if (string.IsNullOrWhiteSpace(root) || !Directory.Exists(root))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Repo Root not found: {root}");
                await InvokeAsync(StateHasChanged);
                return;
            }

            var gitDir = Path.Combine(root, ".git");
            if (!Directory.Exists(gitDir) && !File.Exists(gitDir))
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Not a git repo (missing .git): {root}");
                await InvokeAsync(StateHasChanged);
                return;
            }

            // Full-repo get latest for the BUILD tab button.
            // Per-app project-wise latest still runs later inside BuildApplication.
            var command = "git fetch && git pull --ff-only";

            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Get latest from Code Repo Root...");
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Git: {command}  (cwd: {root})");
            await InvokeAsync(StateHasChanged);

            StartLogFlushTimer();
            var exit = await BuildService.RunCommandAsync(
                root,
                command,
                line =>
                {
                    lock (_pendingLogBuffer)
                    {
                        _pendingLogBuffer.Append(line);
                        if (!line.EndsWith('\n') && !line.EndsWith('\r'))
                            _pendingLogBuffer.AppendLine();
                    }
                },
                CancellationToken.None);

            FlushPendingLogs();

            if (exit != 0)
            {
                AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Get latest failed with exit code {exit}");
                await InvokeAsync(StateHasChanged);
                return;
            }

            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] Get latest OK — loading applications...");

            // Reload apps from configuration (build:apps)
            await LoadApplicationsFromConfigAsync();

            _showApplications = true;
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [INFO] APPLICATIONS ready — {Applications.Count} app(s)");
            await InvokeAsync(StateHasChanged);
        }
        catch (Exception ex)
        {
            AppendLog($"[{DateTime.Now:HH:mm:ss}] [ERROR] Get latest failed: {ex.Message}");
            _showApplications = false;
            await InvokeAsync(StateHasChanged);
        }
        finally
        {
            StopLogFlushTimer();
            _isGettingLatest = false;
            await InvokeAsync(StateHasChanged);
        }
    }

    private Task LoadApplicationsFromConfigAsync()
    {
        // Reuse the same BuildSettings you already loaded in OnInitializedAsync
        // (the object that gave you RepoRoot, OutputRoot, DbRollupExePath, etc.).
        // If that local variable was named differently, use YOUR field/variable name.
        //
        // Common patterns — pick the one that matches your file:
        //   A) field:   private BuildSettings? _buildSettings;
        //   B) local reloaded from config service / IOptions
        //
        // Example A (field set during init):
        var apps = (_buildSettings?.Apps ?? Enumerable.Empty<Application>()).ToList();

        // var buildSettings = Configuration.GetSection("build").Get<BuildSettings>() ?? new();
        // var apps = buildSettings.Apps?.ToList() ?? new List<Application>();

        {
            if (string.IsNullOrWhiteSpace(app.Status))
