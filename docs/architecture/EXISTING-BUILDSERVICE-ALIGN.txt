HOW SetupProjectBuilder uses YOUR existing API
================================================================================

    public class SetupProjectBuilder
    {
        private readonly IBuildService _build;

        public SetupProjectBuilder(IBuildService build) => _build = build;

        public async Task<int> BuildSetupProjectAsync(
            string vdprojPath,
            string workingDir,
            Action<string> log,
            CancellationToken cancellation = default)
        {
            var devenv = _build.ResolveVisualStudioIdePath()
                ?? throw new InvalidOperationException("devenv.com not found");

            // Prefer Process Start on devenv directly (see setup-msi-devenv guide),
            // OR shell via RunCommandAsync if that is what BuildTab already does:
            return await _build.RunCommandAsync(
                workingDir,
                $"\"{devenv}\" \"{vdprojPath}\" /Build Release",
                log,
                cancellation);
        }
    }

    public class BuildOrchestrator
    {
        private readonly IBuildService _build;          // RunCommandAsync + MSBuild path
        private readonly OutputPathService _paths;
        private readonly PublishCopyService _publish;
        private readonly SetupProjectBuilder _setup;

        public async Task BuildApplicationAsync(..., Action<string> log, CancellationToken ct)
        {
            // restore / clean precompiled (as in BuildTab today)

            var msbuild = _build.ResolveMsBuildPath(netFramework: isLegacy);
            // build command may already embed msbuild — keep BuildTab’s command string

            var exit = await _build.RunCommandAsync(workingDir, buildCommand, log, ct);
            if (exit == 0)
                await _publish.CopyArtifactsAsync(...);
            // if Web2ClientBridge setup project:
            //   await _setup.BuildSetupProjectAsync(...);
        }
    }
